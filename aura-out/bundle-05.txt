// supabase/functions/viva-renew/index.ts
//
// STAGE 4: the renewal engine. Called on a schedule (pg_cron, hourly) with a
// secret header. For every Viva subscription that is due it:
//
//   1. reserves the billing period in OUR database   (viva_begin_renewal)
//   2. charges the saved payment method at Viva      (no customer interaction)
//   3. asks Viva to CONFIRM the result before believing it
//   4. applies it atomically                         (viva_apply_renewal)
//
// Safety rules:
//  * At most ONE charge per member per billing period: the order row is the lock.
//  * If we cannot tell whether a charge went through (network cut, timeout,
//    Viva error), the order stays PENDING and is never retried automatically:
//    the money may have moved. The webhook usually settles it; otherwise the
//    admin is alerted.
//  * A declined card starts the grace period (members keep access meanwhile);
//    OUR errors (bad credentials...) are never held against the customer.
//  * Prices come from public.plans at charge time (the admin panel decides).
import { createClient } from "npm:@supabase/supabase-js@^2";
import {
  chargeRecurring,
  loadVivaConfig,
  retrieveTransaction,
  VivaConfigError,
  type VivaConfig,
} from "../_shared/viva.ts";

const EUR_ISO_NUMERIC = "978";
const DEFAULT_BATCH = 25;
const STALE_FROM_MINUTES = 180; // alert once, in the hourly run that first sees an order this old
const STALE_TO_MINUTES = 240;

function json(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json" } });
}

type AdminClient = ReturnType<typeof createClient>;

async function rpc<T = Record<string, unknown>>(admin: AdminClient, name: string, args: Record<string, unknown>): Promise<T> {
  const { data, error } = await admin.rpc(name, args);
  if (error) throw new Error(`${name} failed: ${error.message}`);
  return data as T;
}

async function alertAdmins(admin: AdminClient, title: string, body: string, data: Record<string, unknown>) {
  try {
    await admin.rpc("notify_admins", { p_type: "billing_alert", p_title: title, p_body: body, p_url: "/admin", p_data: data });
  } catch (err) {
    console.error("alertAdmins failed:", err instanceof Error ? err.message : "unknown");
  }
}

/** Constant-time string comparison (the cron secret). */
function safeEqual(a: string, b: string): boolean {
  const enc = new TextEncoder();
  const x = enc.encode(a);
  const y = enc.encode(b);
  let diff = x.length ^ y.length;
  const len = Math.max(x.length, y.length);
  for (let i = 0; i < len; i++) diff |= (x[i] ?? 0) ^ (y[i] ?? 0);
  return diff === 0;
}

interface Summary {
  due: number;
  renewed: number;
  declined: number;
  rejected: number;
  left_pending: number;
  skipped: number;
  anomalies: number;
  ended: number;
}

Deno.serve(async (req: Request): Promise<Response> => {
  if (req.method !== "POST") return json({ error: "method not allowed" }, 405);

  // ---- Only our scheduler may call this. ----
  const expected = Deno.env.get("CRON_SECRET");
  if (!expected) {
    console.error("CRON_SECRET is not set");
    return json({ error: "misconfigured" }, 500);
  }
  if (!safeEqual(req.headers.get("x-cron-secret") ?? "", expected)) return json({ error: "unauthorized" }, 401);

  let cfg: VivaConfig;
  try {
    cfg = loadVivaConfig((name) => Deno.env.get(name));
    if (!cfg.merchantId || !cfg.apiKey) {
      throw new VivaConfigError("Missing Viva secrets: MERCHANT_ID / API_KEY");
    }
  } catch (err) {
    console.error("Viva configuration problem:", err instanceof VivaConfigError ? err.message : "unknown");
    return json({ error: "misconfigured" }, 500);
  }

  const supabaseUrl = Deno.env.get("SUPABASE_URL");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
  if (!supabaseUrl || !serviceKey) {
    console.error("Missing Supabase environment configuration");
    return json({ error: "misconfigured" }, 500);
  }
  const admin = createClient(supabaseUrl, serviceKey, { auth: { persistSession: false } });

  const batch = Math.min(Math.max(Number(Deno.env.get("VIVA_RENEW_BATCH")) || DEFAULT_BATCH, 1), 100);
  const summary: Summary = { due: 0, renewed: 0, declined: 0, rejected: 0, left_pending: 0, skipped: 0, anomalies: 0, ended: 0 };

  try {
    // ---------------- 1. charge what is due ----------------
    const due = await rpc<{ subscription_id: string }[]>(admin, "viva_due_renewals", {
      p_environment: cfg.env,
      p_limit: batch,
    });
    summary.due = Array.isArray(due) ? due.length : 0;

    for (const row of Array.isArray(due) ? due : []) {
      try {
        await renewOne(admin, cfg, row.subscription_id, summary);
      } catch (err) {
        // One member's problem must never stop the others.
        console.error("renewal failed unexpectedly:", err instanceof Error ? err.message : "unknown");
        summary.anomalies += 1;
      }
    }

    // ---------------- 2. end what is over ----------------
    const lapsed = await rpc<{ ended: { user_id: string; reason: string }[] }>(admin, "viva_finish_lapsed", {
      p_environment: cfg.env,
    });
    for (const e of lapsed.ended ?? []) {
      summary.ended += 1;
      const loud = e.reason === "overdue" || e.reason === "no_authorization";
      await alertAdmins(
        admin,
        loud ? "Λήξη συνδρομής Viva (απαιτεί έλεγχο)" : "Λήξη συνδρομής Viva",
        loud
          ? `Η συνδρομή έληξε χωρίς ανανέωση (${e.reason}). Έλεγξε αν υπάρχει πρόβλημα στις χρεώσεις.`
          : `Η συνδρομή έληξε (${e.reason}).`,
        { user_id: e.user_id, kind: `lapsed_${e.reason}` },
      );
    }

    // ---------------- 3. attempts whose outcome we never learned ----------------
    const stale = await rpc<{ order_id: string; user_id: string; amount_cents: number }[]>(admin, "viva_stale_renewals", {
      p_environment: cfg.env,
      p_from_minutes: STALE_FROM_MINUTES,
      p_to_minutes: STALE_TO_MINUTES,
    });
    for (const s of Array.isArray(stale) ? stale : []) {
      await alertAdmins(
        admin,
        "Ανανέωση Viva χωρίς αποτέλεσμα",
        "Μια χρέωση ανανέωσης δεν έχει επιβεβαιωθεί εδώ και ώρες. ΔΕΝ θα ξαναδοκιμαστεί αυτόματα (μπορεί να έχει χρεωθεί ο πελάτης). Έλεγξε τη συναλλαγή στη Viva.",
        { order_id: s.order_id, user_id: s.user_id, amount_cents: s.amount_cents, kind: "renewal_unresolved" },
      );
    }

    console.log("viva-renew run:", JSON.stringify(summary));
    return json({ ok: true, ...summary });
  } catch (err) {
    console.error("viva-renew run failed:", err instanceof Error ? err.message : "unknown");
    return json({ error: "run failed", ...summary }, 500);
  }
});

async function renewOne(admin: AdminClient, cfg: VivaConfig, subscriptionId: string, summary: Summary): Promise<void> {
  const begin = await rpc<{
    result: string;
    order_id?: string;
    user_id?: string;
    plan_id?: string;
    amount_cents?: number;
    initial_transaction_id?: string;
  }>(admin, "viva_begin_renewal", { p_environment: cfg.env, p_subscription_id: subscriptionId });

  if (begin.result === "plan_invalid") {
    summary.anomalies += 1;
    await alertAdmins(admin, "Πρόβλημα ανανέωσης Viva", "Το πλάνο του μέλους δεν έχει έγκυρη τιμή ή νόμισμα. Δεν έγινε χρέωση.", {
      user_id: null, kind: "plan_invalid", subscription_id: subscriptionId,
    });
    return;
  }
  if (begin.result !== "ready" || !begin.order_id || !begin.initial_transaction_id || !begin.amount_cents) {
    summary.skipped += 1; // not due any more / already in progress / no authorization
    return;
  }

  const orderId = begin.order_id;
  const amountCents = begin.amount_cents;

  const charge = await chargeRecurring(cfg, begin.initial_transaction_id, {
    amountCents,
    customerTrns: `AURA ${begin.plan_id ?? ""} - μηνιαία συνδρομή`.trim(),
    merchantTrns: orderId,
  });

  // ---- Outcome we cannot be sure about: do NOT retry, do NOT fail. ----
  if (charge.kind === "unknown") {
    summary.left_pending += 1;
    console.error("renewal outcome unknown:", charge.reason);
    await alertAdmins(admin, "Ανανέωση Viva χωρίς απάντηση", "Η χρέωση ανανέωσης δεν επιβεβαιώθηκε (πρόβλημα επικοινωνίας). ΔΕΝ θα ξαναδοκιμαστεί αυτόματα. Θα ενημερωθείς αν η Viva επιβεβαιώσει την πληρωμή.", {
      order_id: orderId, kind: "renewal_unknown", reason: charge.reason,
    });
    return;
  }

  // ---- OUR side was refused (credentials / request): nothing moved, not the customer's fault. ----
  if (charge.kind === "rejected") {
    summary.rejected += 1;
    await rpc(admin, "viva_fail_renewal", { p_environment: cfg.env, p_order_id: orderId, p_reason: `viva_rejected_${charge.status}`, p_counts: false });
    await alertAdmins(admin, "Η Viva απέρριψε το αίτημα ανανέωσης", `Η Viva απάντησε ${charge.status}. Έλεγξε τα διαπιστευτήρια (Merchant ID / API key) και ότι είναι ενεργοποιημένες οι επαναλαμβανόμενες πληρωμές. Κανένας πελάτης δεν επηρεάστηκε.`, {
      order_id: orderId, kind: "renewal_rejected", http_status: charge.status,
    });
    return;
  }

  // ---- The bank said no. ----
  if (charge.kind === "declined") {
    await failForCustomer(admin, cfg, orderId, `declined_${charge.errorCode ?? "unknown"}`, summary);
    return;
  }

  // ---- Viva gave us a transaction id: remember it, then verify it. ----
  await rpc(admin, "viva_note_renewal_transaction", { p_order_id: orderId, p_transaction_id: charge.transactionId });

  let tx;
  try {
    tx = await retrieveTransaction(cfg, charge.transactionId);
  } catch (err) {
    summary.left_pending += 1;
    console.error("could not verify the renewal with Viva:", err instanceof Error ? err.message : "unknown");
    await alertAdmins(admin, "Ανανέωση Viva χωρίς επιβεβαίωση", "Η χρέωση έγινε αλλά δεν επιβεβαιώθηκε ακόμη από τη Viva. ΔΕΝ θα ξαναδοκιμαστεί αυτόματα.", {
      order_id: orderId, kind: "renewal_unverified",
    });
    return;
  }

  if (tx.statusId === "F") {
    if (tx.merchantTrns !== orderId || tx.amountCents !== amountCents || (tx.currencyCode !== null && tx.currencyCode !== EUR_ISO_NUMERIC)) {
      // The money may have moved but it is not what we asked for. A human decides.
      summary.anomalies += 1;
      summary.left_pending += 1;
      await alertAdmins(admin, "Ανανέωση Viva: ασυμφωνία στοιχείων", "Η Viva επιβεβαίωσε πληρωμή που δεν ταιριάζει με την παραγγελία μας. Δεν έγινε ενεργοποίηση.", {
        order_id: orderId, transaction_id: charge.transactionId, kind: "renewal_mismatch",
        expected_cents: amountCents, received_cents: tx.amountCents,
      });
      return;
    }

    const applied = await rpc<{ result: string }>(admin, "viva_apply_renewal", {
      p_environment: cfg.env,
      p_merchant_trns: orderId,
      p_transaction_id: tx.transactionId,
      p_amount_cents: tx.amountCents,
      p_paid_at: tx.insDate,
    });

    if (applied.result === "renewed" || applied.result === "already_processed") {
      summary.renewed += 1;
      return;
    }
    summary.anomalies += 1;
    await alertAdmins(admin, "Πρόβλημα ανανέωσης Viva", `Η πληρωμή επιβεβαιώθηκε αλλά δεν εφαρμόστηκε (${applied.result}). Έλεγξε τη συναλλαγή.`, {
      order_id: orderId, transaction_id: charge.transactionId, kind: applied.result,
    });
    return;
  }

  if (tx.statusId === "E") {
    await failForCustomer(admin, cfg, orderId, `viva_status_${tx.statusId}`, summary);
    return;
  }

  // Any other status is not final: leave the order pending and let the webhook decide.
  summary.left_pending += 1;
  await alertAdmins(admin, "Ανανέωση Viva σε αναμονή", `Η συναλλαγή έχει κατάσταση ${tx.statusId} (όχι τελική). Αναμένεται ειδοποίηση από τη Viva.`, {
    order_id: orderId, transaction_id: charge.transactionId, kind: "renewal_not_final",
  });
}

async function failForCustomer(admin: AdminClient, cfg: VivaConfig, orderId: string, reason: string, summary: Summary) {
  summary.declined += 1;
  const r = await rpc<{ result: string; attempts?: number; user_id?: string; grace_until?: string; will_retry?: boolean }>(
    admin,
    "viva_fail_renewal",
    { p_environment: cfg.env, p_order_id: orderId, p_reason: reason, p_counts: true },
  );
  // One alert per member for the first failure; the end of the grace period alerts again.
  if (r.result === "failure_recorded" && (r.attempts ?? 1) <= 1) {
    await alertAdmins(admin, "Αποτυχία ανανέωσης συνδρομής Viva", "Η κάρτα του μέλους απορρίφθηκε. Το μέλος κρατά πρόσβαση στην περίοδο χάριτος και θα γίνουν νέες προσπάθειες.", {
      user_id: r.user_id ?? null, kind: "renewal_declined", grace_until: r.grace_until ?? null,
    });
  }
}

export function classify(p: WebhookPayload, opts: { bookTable: string }): EventKind | null {
  const rec = p.record ?? {};
  const wasPaid = p.old_record?.status === "paid";
  const nowPaid = rec.status === "paid";
  const becamePaid = nowPaid && (p.type === "INSERT" || !wasPaid);

  switch (p.table) {
    case "contact_messages":
      if (p.type !== "INSERT") return null;
      return rec.kind === "appointment" ? "contact_appointment" : "contact_message";
    case "seminar_bookings":
      return p.type === "INSERT" ? "seminar_booking" : null; // κράτηση σεμιναρίου επί τόπου
    case "online_seminar_bookings":
      return becamePaid ? "online_seminar_paid" : null; // πληρωμένη κράτηση online σεμιναρίου
    default:
      if (p.table === opts.bookTable) return becamePaid ? "book_sold" : null; // πουλήθηκε βιβλίο
      return null;
  }
}

export const esc = (v: unknown): string =>
  String(v ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");

const str = (v: unknown): string => (v === null || v === undefined ? "" : String(v).trim());
const isEmail = (v: string) => /^[^@\s,;<>]+@[^@\s,;<>]+\.[^@\s,;<>]+$/.test(v);

function money(cents: unknown, currency: unknown = "eur"): string {
  const n = Number(cents);
  if (!Number.isFinite(n)) return "—";
  try {
    return new Intl.NumberFormat("el-GR", { style: "currency", currency: str(currency).toUpperCase() || "EUR" }).format(n / 100);
  } catch {
    return `${(n / 100).toFixed(2)} €`;
  }
}

function dateTime(iso: unknown): string {
  const s = str(iso);
  if (!s) return "—";
  const d = new Date(s);
  if (Number.isNaN(d.getTime())) return s;
  return d.toLocaleString("el-GR", { timeZone: "Europe/Athens", dateStyle: "full", timeStyle: "short" });
}

function dateOnly(v: unknown): string {
  const s = str(v);
  if (!s) return "—";
  const d = new Date(`${s}T12:00:00`);
  if (Number.isNaN(d.getTime())) return s;
  return d.toLocaleDateString("el-GR", { timeZone: "Europe/Athens", weekday: "long", day: "numeric", month: "long", year: "numeric" });
}

type Line = { label: string; value: string; href?: string };

const clean = (lines: (Line | null | false)[]): Line[] =>
  lines.filter((l): l is Line => !!l && l.value !== "" && l.value !== "—");

function render(spec: Spec, siteUrl?: string): Mail {
  const base = str(siteUrl).replace(/\/+$/, "");
  const url = spec.link && base ? `${base}${spec.link.path}` : "";

  const rowsHtml = spec.lines
    .map((l) => {
      const val = l.href
        ? `<a href="${esc(l.href)}" style="color:#111;text-decoration:underline">${esc(l.value)}</a>`
        : esc(l.value).replace(/\n/g, "<br>");
      return `<tr><td style="padding:9px 14px 9px 0;color:#6b7280;font-size:13px;vertical-align:top;white-space:nowrap">${esc(l.label)}</td><td style="padding:9px 0;color:#111;font-size:14px;font-weight:500;vertical-align:top">${val}</td></tr>`;
    })
    .join("");

  const blocksHtml = (spec.blocks ?? [])
    .map(
      (b) =>
        `<div style="margin-top:18px"><div style="font-size:11px;letter-spacing:.14em;text-transform:uppercase;color:#6b7280;margin-bottom:6px">${esc(b.label)}</div><div style="background:#f4f4f5;border-radius:10px;padding:14px 16px;font-size:14px;line-height:1.6;color:#111;white-space:pre-wrap">${esc(b.text)}</div></div>`,
    )
    .join("");

  const button = url
    ? `<div style="margin-top:24px"><a href="${esc(url)}" style="display:inline-block;background:#111;color:#fff;text-decoration:none;font-size:13px;font-weight:600;padding:11px 20px;border-radius:999px">${esc(spec.link!.label)}</a></div>`
    : "";

  const html = `<!doctype html><html lang="el"><body style="margin:0;padding:24px;background:#eef0f3;font-family:-apple-system,Segoe UI,Roboto,Helvetica,Arial,sans-serif">
<div style="max-width:560px;margin:0 auto;background:#fff;border-radius:16px;overflow:hidden;border:1px solid #e5e7eb">
<div style="background:#0b0b0c;padding:18px 24px"><span style="color:#fff;font-size:15px;font-weight:700;letter-spacing:.32em">AURA</span><span style="color:#9ca3af;font-size:11px;letter-spacing:.2em;text-transform:uppercase;margin-left:12px">${esc(spec.badge)}</span></div>

<h1 style="margin:0;font-size:20px;line-height:1.3;color:#111">${esc(spec.title)}</h1>
${spec.lead ? `<p style="margin:8px 0 0;font-size:14px;color:#4b5563">${esc(spec.lead)}</p>` : ""}
<table role="presentation" cellpadding="0" cellspacing="0" style="margin-top:18px;width:100%;border-top:1px solid #eee">${rowsHtml}</table>
${blocksHtml}${button}
</div></div>
<p style="max-width:560px;margin:14px auto 0;text-align:center;font-size:11px;color:#9ca3af">Αυτόματη ειδοποίηση από το AURA. Μπορείς να απαντήσεις απευθείας σε αυτό το email.</p>
</body></html>`;

  const text = [
    spec.title,
    spec.lead ?? "",
    "",
    ...spec.lines.map((l) => `${l.label}: ${l.value}`),
    ...(spec.blocks ?? []).flatMap((b) => ["", `${b.label}:`, b.text]),
    url ? `\n${spec.link!.label}: ${url}` : "",

    .filter((s, i, a) => !(s === "" && a[i - 1] === ""))
    .join("\n")
    .trim();

  return { subject: spec.subject.replace(/[\r\n]+/g, " ").trim(), html, text, replyTo: spec.replyTo };
}

// src/lib/create-viva-checkout.ts
import { supabase } from "@/lib/supabase";
import type { PlanId } from "@/lib/plans";

/** The only places a customer may be sent to pay (demo and live Smart Checkout). */
const ALLOWED_CHECKOUT_PREFIXES = [
  "https://demo.vivapayments.com/web/checkout?ref=",
  "https://www.vivapayments.com/web/checkout?ref=",
];

export function isVivaCheckoutUrl(url: string): boolean {
  return ALLOWED_CHECKOUT_PREFIXES.some((prefix) => url.startsWith(prefix));
}

/**
 * Asks the create-viva-checkout Edge Function for a payment page and returns
 * its URL. The browser sends ONLY the plan id: price, user and eligibility are
 * all decided on the server. Throws an Error with a readable message.
 */
export async function createVivaCheckout(planId: PlanId): Promise<string> {
  const { data, error } = await supabase.functions.invoke<{
    redirectUrl?: string;
    error?: string;
  }>("create-viva-checkout", { body: { planId } });

  if (error) {
    const res = (error as { context?: Response }).context;
    if (res && typeof res.json === "function") {
      try {
        const payload = (await res.json()) as { error?: string };
        if (payload?.error) throw new Error(payload.error);
      } catch (parseErr) {
        if (parseErr instanceof Error && parseErr.message) throw parseErr;
      }
    }
    throw new Error(error.message || "Η πληρωμή δεν μπόρεσε να ξεκινήσει.");
  }

  if (data?.error) throw new Error(data.error);
  const url = data?.redirectUrl;
  // Never follow an address we do not recognise.
  if (!url || !isVivaCheckoutUrl(url)) throw new Error("Δεν δημιουργήθηκε σελίδα πληρωμής.");
  return url;
}

export function validateBillingProfile(p: BillingProfile): ProfileErrors {
  const e: ProfileErrors = {};
  const len = (v: string) => v.trim().length;
  if (len(p.full_name) < 2 || len(p.full_name) > 120) e.full_name = MESSAGES.full_name;
  if (!/^[0-9+()\s.-]{6,40}$/.test(p.phone.trim())) e.phone = MESSAGES.phone;
  if (len(p.address_line1) < 3 || len(p.address_line1) > 200) e.address_line1 = MESSAGES.address_line1;
  if (len(p.city) < 2 || len(p.city) > 100) e.city = MESSAGES.city;
  if (len(p.postal_code) < 2 || len(p.postal_code) > 20) e.postal_code = MESSAGES.postal_code;
  if (len(p.country) < 2 || len(p.country) > 100) e.country = MESSAGES.country;
  if (p.wants_invoice) {
    if (len(p.company_name) < 1 || len(p.company_name) > 200) e.company_name = MESSAGES.company_name;
    if (!/^[A-Za-z0-9]{8,14}$/.test(p.vat_number.replace(/[\s.-]/g, ""))) e.vat_number = MESSAGES.vat_number;
    if (len(p.tax_office) < 1 || len(p.tax_office) > 120) e.tax_office = MESSAGES.tax_office;
  }
  return e;
}

// src/lib/viva-return.ts
//
// What Viva puts in the URL when it sends a customer back to the site:
//   success:  ?t=<transaction id>&s=<order code>&lang=el-GR&eventId=0
//   refused / cancelled:  ?s=<order code>&lang=el-GR&eventId=<code>      (NO t)
// These parameters are never trusted: the checkout return page hands them to
// the server, which asks Viva directly what happened.
import { supabase } from "@/lib/supabase";

const UUID = /^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$/;

export interface VivaReturnParams {
  /** Transaction id: only present when the payment went through. */
  t: string | null;
  /** Viva order code. */
  s: string;
  /** Viva event id. */
  eventId: number | null;
}

/** Reads the parameters if (and only if) the URL looks like a return from Viva. */
export function readVivaReturnParams(search: string | URLSearchParams): VivaReturnParams | null {
  const sp = typeof search === "string" ? new URLSearchParams(search) : search;
  const s = sp.get("s");
  if (!s || !/^\d{6,20}$/.test(s)) return null;

  const tRaw = sp.get("t");
  const t = tRaw && UUID.test(tRaw) ? tRaw.toLowerCase() : null;
  const eRaw = sp.get("eventId");
  const eventId = eRaw !== null && /^\d{1,7}$/.test(eRaw) ? Number(eRaw) : null;

  // A real Viva return has either the transaction id or an event id.
  if (t === null && eventId === null) return null;
  return { t, s, eventId };
}

export type VivaPageStatus = "paid" | "pending" | "failed" | "unknown";

export interface VivaVerifyResult {
  /** subscription_initial | subscription_renewal | subscription_upgrade | seminar | book */
  kind: string;
  status: VivaPageStatus;
  label: string | null;
  amountCents: number | null;
  eventId: number | null;
}

/** Asks the server to confirm the payment with Viva (safe to call repeatedly). */
export async function verifyVivaPayment(p: VivaReturnParams): Promise<VivaVerifyResult> {
  const { data, error } = await supabase.functions.invoke<Partial<VivaVerifyResult> & { error?: string }>("verify-viva-payment", {
    body: { t: p.t, s: p.s, eventId: p.eventId },
  });

  if (error) {
    const res = (error as { context?: Response }).context;
    // An order we do not know is a definite answer, not a glitch.
    if (res && res.status === 404) return { kind: "unknown", status: "unknown", label: null, amountCents: null, eventId: p.eventId };
    throw new Error(error.message || "Η επαλήθευση της πληρωμής δεν ήταν δυνατή.");
  }
  const status = data?.status;
  return {
    kind: data?.kind ?? "unknown",
    status: status === "paid" || status === "pending" || status === "failed" ? status : "unknown",
    label: data?.label ?? null,
    amountCents: typeof data?.amountCents === "number" ? data.amountCents : null,
    eventId: typeof data?.eventId === "number" ? data.eventId : p.eventId,
  };
}

/** Plain-language explanation shown when a payment did not go through. */
export function failureMessage(eventId: number | null | undefined): string {
  const code = typeof eventId === "number" ? ` (κωδικός Viva: ${eventId})` : "";
  return `Η πληρωμή δεν ολοκληρώθηκε${code}. Δεν χρεώθηκες. Έλεγξε τα στοιχεία της κάρτας, δοκίμασε άλλη κάρτα ή ξαναπροσπάθησε.`;
}
