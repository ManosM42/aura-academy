function RootComponent() {
  const { queryClient } = Route.useRouteContext();

  // Viva sends customers back to the "Success" / "Failure" URL set in the Viva
  // dashboard (it may be /pricing or any other page) with ?t=..&s=..&eventId=..
  // Wherever they land, forward them to the page that confirms the payment.
  useEffect(() => {
    const here = window.location;
    if (here.pathname.startsWith("/checkout/return")) return;
    if (readVivaReturnParams(here.search)) {
      window.location.replace(`/checkout/return${here.search}`);
    }
  }, []);

  return (
    <QueryClientProvider client={queryClient}>
      <I18nProvider>
        <CookieConsentProvider>
        <AuraWidget />
          <CallProvider>

function indexAvailability(rows) {
	const map = /* @__PURE__ */ new Map();
	for (const r of rows) {
		const day = map.get(r.seminar_date) ?? {
			dayBlocked: false,
			people: 0,
			priv: false
		};
		if (r.blocked) day.dayBlocked = true;

function rowsFromBookings(bookings, blocked) {
	const rows = [];
	for (const b of bookings) {
		if (b.status === "cancelled") continue;
		rows.push({
			seminar_date: b.seminar_date,
			slot: b.slot,
			group_people: b.format === "group" ? b.participants : 0,
			has_private: b.format === "private",
			blocked: false
		});
	}
	for (const d of blocked) rows.push({
		seminar_date: d.blocked_date,
		slot: d.slot,
		group_people: 0,
		has_private: false,
		blocked: true
	});
	return rows;
}
function getDayAvailability(day, format, participants) {
	if (day?.dayBlocked) return {
		status: "blocked",
		remaining: 0
	};
	const taken = day?.priv ? 8 : day?.people ?? 0;
	const remaining = Math.max(0, 8 - taken);
	if (format === "private") return taken > 0 ? {
		status: "full",
		remaining: 0
	} : {
		status: "open",
		remaining: 8
	};
	if (remaining < participants) return {
		status: "full",
		remaining
	};
	return {
		status: remaining <= 3 ? "few" : "open",
		remaining
	};
}

// src/lib/access.ts
import type { SupabaseClient } from "@supabase/supabase-js";
import { db } from "@/lib/db";
import { isStaffRole } from "@/lib/queries";

// New functions/tables are not in the generated types yet.
const sb = db as unknown as SupabaseClient;

/**
 * Pages anyone can open, even without a plan.
 * "/login" and "/checkout" MUST stay here: people sign in and pay through them.
 * Add any other public page here (matches the path itself and everything under it).
 */
export const PUBLIC_PATHS = [
  "/",
  "/pricing",
  "/terms",
  "/contact",
  "/about",
  "/ourteam",
  "/login",
  "/checkout",
  // Anyone can buy these without a plan: the book and the seminars.
  "/book",
  "/seminars",
  "/online-seminars",
];

export function isPublicPath(pathname: string): boolean {
  const lower = pathname.toLowerCase();
  const p = lower.length > 1 ? lower.replace(/\/+$/, "") : lower;
  return PUBLIC_PATHS.some((base) =>
    base === "/" ? p === "/" : p === base || p.startsWith(`${base}/`),
  );
}

const ACCESS_STATUSES = ["active", "trialing"];

export interface AccessState {
  /** Staff, or an active/trialing subscription. */
  hasAccess: boolean;
  isStaff: boolean;
  /** The user has (or had) a subscription row, e.g. an expired one. */
  hadPlan: boolean;
}

/** Throws on a network/database error (callers must NOT treat that as "no plan"). */
export async function fetchAccess(userId: string): Promise<AccessState> {
  const { data: profile, error: profileError } = await sb
    .from("profiles")
    .select("role")
    .eq("id", userId)
    .maybeSingle();
  if (profileError) throw new Error(profileError.message);

  if (isStaffRole(String((profile as { role?: unknown } | null)?.role ?? "student"))) {
    return { hasAccess: true, isStaff: true, hadPlan: false };
  }

  const { data: subs, error: subsError } = await sb
    .from("subscriptions")
    .select("status")
    .eq("user_id", userId);
  if (subsError) throw new Error(subsError.message);

  const rows = (subs ?? []) as { status: string }[];
  return {
    hasAccess: rows.some((r) => ACCESS_STATUSES.includes(r.status)),
    isStaff: false,
    hadPlan: rows.length > 0,
  };
}

/** Removes the caller's dead plan rows (never touches active ones). Payments are kept. */
export async function cleanupDeadSubscriptions(): Promise<void> {
  const { error } = await sb.rpc("cleanup_my_dead_subscriptions");
  if (error) console.error("[access] cleanup_my_dead_subscriptions failed:", error.message);
}

const TIMEOUT_MS = 12000;

interface AuthCallbackSearch {
  plan?: PlanId;
  /** Το index signature είναι σκόπιμο: το validateSearch ΔΕΝ πρέπει να
   *  πετάξει τα ?code / ?error του Supabase, αλλιώς χάνεται το PKCE code. */
  [key: string]: unknown;
}

export const Route = createFileRoute("/auth/callback")({
  validateSearch: (search: Record<string, unknown>): AuthCallbackSearch => ({
    ...search,
    plan: isPlanId(search["plan"]) ? search["plan"] : undefined,
  }),
  head: () => ({ meta: [{ title: "AURA — Sign in" }] }),
  component: AuthCallbackPage,
});

function AuthCallbackPage() {
  const navigate = useNavigate();
  const { user, loading } = useAuth();
  const { plan: searchPlan } = Route.useSearch();
  const { t } = useI18n();
  const [failure, setFailure] = useState<string | null>(null);
  const [timedOut, setTimedOut] = useState(false);

  // Τα σφάλματα OAuth έρχονται είτε στο query string είτε στο hash fragment.
  useEffect(() => {
    if (typeof window === "undefined") return;
    const search = new URLSearchParams(window.location.search);
    const hash = new URLSearchParams(window.location.hash.replace(/^#/, ""));
    const description =
      search.get("error_description") ??
      hash.get("error_description") ??
      search.get("error") ??
      hash.get("error");
    if (description) setFailure(description.replace(/\+/g, " "));
  }, []);

  // Επιτυχία: πρώτα το URL, μετά το αποθηκευμένο intent. Αυτό σταματά το
  // σιωπηλό fall-through στο /dashboard όταν το storage είναι άδειο.
  useEffect(() => {
    if (failure || loading || !user) return;
    const plan = searchPlan ?? consumeCheckoutIntent();
    if (searchPlan) clearCheckoutIntent(); // πέτα τυχόν παλιό leftover
    if (plan) {
      navigate({ to: "/checkout", search: { plan }, replace: true });
    } else {
      navigate({ to: "/dashboard", replace: true });
    }
  }, [failure, loading, user, searchPlan, navigate]);

  // Δικλείδα: αν το session δεν έρθει ποτέ, μην αφήνεις τον χρήστη σε loader.
  useEffect(() => {
    if (failure || user) return;
    const timer = window.setTimeout(() => setTimedOut(true), TIMEOUT_MS);
    return () => window.clearTimeout(timer);
  }, [failure, user]);

  const blocked = failure !== null || timedOut;

  return (
    <main
      className="mx-auto max-w-lg px-5 pb-24 pt-32 text-center sm:pt-40"
      aria-busy={!blocked}

          </p>
          <h1 className="mt-5 text-3xl font-semibold tracking-tight text-neutral-100">
            {failure ? t("auth.errorTitle") : t("auth.timeoutTitle")}
          </h1>
          <p className="mt-4 text-sm leading-relaxed text-neutral-400" role="alert">
            {failure ?? t("auth.timeoutBody")}
          </p>
          <div className="mt-8 flex flex-col items-center gap-3 sm:flex-row sm:justify-center">
            <Link to="/login" search={searchPlan ? { plan: searchPlan } : {}}>
              <ChromeButton type="button">{t("auth.backToLogin")}</ChromeButton>
            </Link>
