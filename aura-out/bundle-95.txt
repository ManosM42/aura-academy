var ROLE_RANK = {
	student: 10,
	educator: 20,
	senior_educator: 30,
	content_manager: 30,
	operations: 35,
	admin: 40,
	super_admin: 50
};
/** Ελάχιστος βαθμός για τα εργαλεία του team (Review Queue). */
var STAFF_RANK = ROLE_RANK.educator;
/** Ελάχιστος βαθμός για Admin Panel & Inbox. */
var ADMIN_RANK = ROLE_RANK.admin;
/** Άγνωστος, κενός ή null ρόλος → 0. Fail closed. */
function roleRank(role) {
	if (!role) return 0;
	const key = role.trim().toLowerCase().replace(/[\s-]+/g, "_");
	return key in ROLE_RANK ? ROLE_RANK[key] : 0;
}
/** Educator και πάνω. */
function isStaffRole(role) {
	return roleRank(role) >= STAFF_RANK;
}
/** Admin και πάνω — Admin Panel και Inbox. */
function isAdminRole(role) {
	return roleRank(role) >= ADMIN_RANK;
}
/** Ρητό κατώφλι, όταν θέλεις κάτι ενδιάμεσο. */
function hasAtLeastRole(role, minimum) {
	return roleRank(role) >= ROLE_RANK[minimum];
}
var CONTENT_ROLES = [
	"content_manager",
	"operations",
	"admin",
	"super_admin"
];
function isContentRole(role) {
	if (!role) return false;
	const key = role.trim().toLowerCase().replace(/[\s-]+/g, "_");
	return CONTENT_ROLES.includes(key);
}
//#endregion
export { isStaffRole as i, isAdminRole as n, isContentRole as r, hasAtLeastRole as t };
