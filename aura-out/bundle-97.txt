// src/components/aura/AuraIntro.tsx
//
// First-visit intro for the AURA platform.
// Plays once per browser, the first time someone enters the site on any page.
// Clear AURA_INTRO_KEY to replay.

import {
  useCallback,
  useEffect,
  useLayoutEffect,
  useRef,
  useState,
} from "react";
import { motion } from "motion/react";
import welcomeVideo from "@/assets/welcome.mp4";

export const AURA_INTRO_KEY = "aura:intro-seen";

type Phase = "playing" | "outro" | "reveal" | "done";

const EASE = [0.22, 1, 0.36, 1] as const;
const useIsoLayoutEffect =
  typeof window !== "undefined" ? useLayoutEffect : useEffect;

const LOAD_TIMEOUT_MS = 8000;

function hasSeenIntro(): boolean {
  try {
    return localStorage.getItem(AURA_INTRO_KEY) !== null;
  } catch {
    return false; // storage blocked: treat as first visit
  }
}

function markIntroSeen() {
  try {
    localStorage.setItem(AURA_INTRO_KEY, "1");
  } catch {
    /* ignore */
  }
}

interface AuraIntroProps {
  onComplete?: () => void;
}

export default function AuraIntro({ onComplete }: AuraIntroProps) {
  const [phase, setPhase] = useState<Phase>("playing");
  const [ready, setReady] = useState(false);

  const videoRef = useRef<HTMLVideoElement>(null);
  const finishing = useRef(false);

  const onCompleteRef = useRef(onComplete);
  useEffect(() => {
    onCompleteRef.current = onComplete;
  }, [onComplete]);

  // Returning visitor: skip before the first paint.
  useIsoLayoutEffect(() => {
    if (hasSeenIntro()) {
      setPhase("done");
      onCompleteRef.current?.();
    }
  }, []);

  // Lock page scroll while the intro is up.
  useEffect(() => {
    if (phase === "done") return;
    const prev = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    return () => {
      document.body.style.overflow = prev;
    };
  }, [phase === "done"]); // eslint-disable-line react-hooks/exhaustive-deps

  const finish = useCallback(() => {
    if (finishing.current) return;
    finishing.current = true;
    markIntroSeen(); // also mark here, so a finished intro never replays
    setPhase("outro");
    window.setTimeout(() => setPhase("reveal"), 900);
    window.setTimeout(() => {
      setPhase("done");
      onCompleteRef.current?.();
    }, 2100);
  }, []);

  // Start playing on mount, but never for returning visitors.
  useEffect(() => {
    if (phase !== "playing" || hasSeenIntro()) return;
    const v = videoRef.current;
    if (!v) return;

    v.muted = true; // autoplay is only allowed when muted
    v.play().catch(finish);

    const timeout = window.setTimeout(() => {
      if (!v || v.readyState < 3 || v.paused) finish();
    }, LOAD_TIMEOUT_MS);
    return () => window.clearTimeout(timeout);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  if (phase === "done") return null;

  return (
    <motion.div
      role="dialog"
      aria-label="Welcome to AURA"
      initial={{ opacity: 1 }}
      animate={{ opacity: phase === "reveal" ? 0 : 1 }}
      transition={{ duration: 1.2, ease: EASE }}
      style={{ pointerEvents: phase === "reveal" ? "none" : "auto" }}
      className="fixed inset-0 z-[100] overflow-hidden bg-black"
    >
      <motion.div
        initial={{ opacity: 0 }}
        animate={{ opacity: phase === "playing" && ready ? 1 : 0 }}
        transition={{
          duration: phase === "outro" ? 0.9 : 0.8,
          ease: "easeInOut",
        }}
        className="pointer-events-none absolute inset-0 bg-black"
      >
        <video
          ref={videoRef}
          src={welcomeVideo}
          autoPlay
          muted
          playsInline
          preload="auto"
          disablePictureInPicture
          className="h-full w-full bg-black object-contain"
          onPlaying={() => {
            setReady(true);
            markIntroSeen();
          }}
          onLoadedMetadata={(e) => {
            if (e.currentTarget.videoWidth === 0) finish();
          }}
          onEnded={finish}
          onError={finish}
        />
      </motion.div>
    </motion.div>
  );
}

import { useEffect, useRef, useState } from "react";
import { AnimatePresence, motion } from "motion/react";
import auraLogo from "@/assets/aura.jpg";

interface AuraLoaderProps {
  onComplete?: () => void;
}

export default function AuraLoader({ onComplete }: AuraLoaderProps) {
  // Step sequence:
  // 1: "Welcome" (1 sec)
  // 2: "To" (1 sec)
  // 3: aura.jpg (2 sec)
  // 4: Solid black screen fades out to reveal the landing page
  // "done": Unmounts component completely
  const [step, setStep] = useState<1 | 2 | 3 | 4 | "done">(1);

  const onCompleteRef = useRef(onComplete);
  useEffect(() => {
    onCompleteRef.current = onComplete;
  }, [onComplete]);

  useEffect(() => {
    let timer: number;

    if (step === 1) {
      // Step 1 ("Welcome"): 1 second
      timer = window.setTimeout(() => setStep(2), 1000);
    } else if (step === 2) {
      // Step 2 ("To"): 1 second
      timer = window.setTimeout(() => setStep(3), 1000);
    } else if (step === 3) {
      // Step 3 (aura.jpg): 2 seconds
      timer = window.setTimeout(() => setStep(4), 2000);
    } else if (step === 4) {
      // Step 4 (Black screen fade out to main site): 1.2 seconds
      timer = window.setTimeout(() => {
        setStep("done");
        onCompleteRef.current?.();
      }, 1200);
    }

    return () => window.clearTimeout(timer);
  }, [step]);

  if (step === "done") return null;

  const chromeTextStyle: React.CSSProperties = {
    backgroundImage:
      "linear-gradient(to bottom, #ffffff 0%, #e0e0e0 30%, #7a7a7a 49%, #292929 51%, #a3a3a3 60%, #ffffff 100%)",

    filter:
      "drop-shadow(0px 4px 12px rgba(255,255,255,0.15)) drop-shadow(0px 1px 2px rgba(255,255,255,0.3))",
  };

  return (
    <motion.div
      key="aura-loader-black-screen"
      initial={{ opacity: 1 }}
      animate={{ opacity: step === 4 ? 0 : 1 }}

      className="fixed inset-0 z-[100] flex items-center justify-center bg-black overflow-hidden pointer-events-none"

      <AnimatePresence mode="wait">
        {step === 1 && (
          <motion.h1
            key="text-welcome"
            style={chromeTextStyle}
            initial={{ opacity: 0, scale: 0.92, filter: "blur(16px)" }}
            animate={{ opacity: 1, scale: 1.02, filter: "blur(0px)" }}
            exit={{ opacity: 0, scale: 1.08, filter: "blur(16px)" }}
            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
            className="notranslate text-5xl sm:text-7xl md:text-8xl font-extrabold uppercase tracking-[0.35em] select-none"
          >
            Welcome
          </motion.h1>
        )}

        {step === 2 && (
          <motion.h1
            key="text-to"
            style={chromeTextStyle}
            initial={{ opacity: 0, scale: 0.92, filter: "blur(16px)" }}
            animate={{ opacity: 1, scale: 1.02, filter: "blur(0px)" }}
            exit={{ opacity: 0, scale: 1.08, filter: "blur(16px)" }}
            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
            className="notranslate text-5xl sm:text-7xl md:text-8xl font-extrabold uppercase tracking-[0.35em] select-none"

// Gentle, small snowflakes for the Christmas vibe. Uses the `falling-snow` package
// (npm i falling-snow). The snow never blocks clicks and respects "reduce motion".
import { useEffect, useState } from "react";
import { Snowfall } from "falling-snow/react";
import { isWinter } from "falling-snow";

interface ChristmasSnowProps {
  /** Number of flakes on desktop (phones get ~60% of this). */
  count?: number;
  /**
   * Layer of the snow. 40 keeps it above the page but under the intro video and dialogs.
   * Use 9999 if you also want snow over the navbar.
   */
  zIndex?: number;
  /** Only snow in December, January and February (hides itself the rest of the year). */
  onlyInWinter?: boolean;
}

export default function ChristmasSnow({ count = 110, zIndex = 40, onlyInWinter = false }: ChristmasSnowProps) {
  const [flakes, setFlakes] = useState<number | null>(null);

  // Client only: decide the amount after mount (no server/client mismatch).
  useEffect(() => {
    const phone = window.matchMedia("(max-width: 640px)");
    const apply = () => setFlakes(phone.matches ? Math.round(count * 0.6) : count);
    apply();
    phone.addEventListener("change", apply);
    return () => phone.removeEventListener("change", apply);
  }, [count]);

  if (flakes === null) return null;
  if (onlyInWinter && !isWinter()) return null;

  return (
    <Snowfall
      fullScreen
      flake="mixed" // real crystal shapes, from tiny soft dots to detailed stars
      count={flakes}
      size={[3, 9]} // a bit bigger than before, still delicate
      speed={0.9}
      wind={0.35} // a light breeze
      sway={1.5} // natural side-to-side flutter
      spin={0.8}
      depth={1} // full depth of field: far flakes small, slow and blurry; near ones sharp
      opacity={[0.25, 0.95]}
      colors={["#ffffff", "#f4f8ff", "#e3eeff", "#d6e6ff"]}
      zIndex={zIndex}
    />
  );
}

import { useEffect, useRef } from "react";

/**
 * A barely-visible grayscale radial light that follows the pointer,
 * evoking reflected light on polished metal. Not an RGB gaming glow.
 * Disabled on touch devices and when reduced motion is preferred.
 */
export default function ChromeCursor() {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const coarse = window.matchMedia("(pointer: coarse)").matches;
    if (reduced || coarse) return;

    let raf = 0;
    let x = window.innerWidth / 2;
    let y = window.innerHeight / 2;

    const render = () => {
      el.style.transform = `translate3d(${x - 300}px, ${y - 300}px, 0)`;
      raf = requestAnimationFrame(render);
    };

    const onMove = (e: PointerEvent) => {
      x = e.clientX;
      y = e.clientY;
    };

    window.addEventListener("pointermove", onMove, { passive: true });
    raf = requestAnimationFrame(render);

    return () => {
      window.removeEventListener("pointermove", onMove);
      cancelAnimationFrame(raf);
    };
  }, []);

  return (
    <div
      ref={ref}
      aria-hidden="true"
      className="pointer-events-none fixed left-0 top-0 z-[2] h-[600px] w-[600px] rounded-full"
      style={{
        background:
          "radial-gradient(circle, rgba(255,255,255,0.045) 0%, rgba(191,192,194,0.02) 35%, transparent 62%)",
      }}
    />
  );
}

// mouthWarp.ts
// Κίνηση στόματος ανά pixel (όχι λωρίδες): το πάνω χείλος ανεβαίνει, το κάτω χείλος και το σαγόνι
// κατεβαίνουν, και στο άνοιγμα ζωγραφίζεται ρεαλιστικό εσωτερικό (κοιλότητα, δόντια, γλώσσα).
// Δεν εξαρτάται από DOM/React, άρα δοκιμάζεται σε node.

export type MouthSpec = {
  /** Κέντρο στόματος και γραμμή ανάμεσα στα χείλη, κλίμακα 0..1 της εικόνας */
  readonly x: number;
  readonly y: number;
  /** μισό πλάτος χειλιών (γωνία–κέντρο) ως ποσοστό του πλάτους */
  readonly halfW: number;
  /** πόσο κάτω από τη γραμμή των χειλιών φτάνει το σαγόνι (ποσοστό ύψους) */
  readonly jaw: number;
  /** ύψος πάνω χείλους (ποσοστό ύψους) */
  readonly lip: number;
  /** μέγιστο άνοιγμα στόματος (ποσοστό ύψους) */
  readonly maxOpen: number;
};

export type MouthRuntime = {
  rx: number;
  ry: number;
  w: number;
  h: number;
  orig: Uint8ClampedArray;
  /** Όλα σε τοπικές συντεταγμένες της περιοχής (pixels) */
  mx: number;
  my: number;
  hw0: number;
  jaw: number;
  lip: number;
  maxOpen: number;
  margin: number;
  s: number;
};

const REF_HALF_W = 79; // μισό πλάτος χειλιών στη φωτογραφία όπου ρυθμίστηκαν τα μεγέθη των δοντιών

const clamp = (v: number, a: number, b: number) => (v < a ? a : v > b ? b : v);
const clamp01 = (v: number) => clamp(v, 0, 1);
const sstep = (t: number) => {
  const x = clamp01(t);
  return x * x * (3 - 2 * x);
};
const mix = (a: number, b: number, t: number) => a + (b - a) * t;

export function buildMouth(
  spec: MouthSpec,
  W: number,
  H: number,
  getRegion: (x: number, y: number, w: number, h: number) => Uint8ClampedArray,
): MouthRuntime {
  const mxA = spec.x * W;
  const myA = Math.round(spec.y * H);
  const hw0 = spec.halfW * W;
  const jaw = spec.jaw * H;
  const lip = spec.lip * H;
  const margin = hw0 * 1.5;

  const rx = Math.max(0, Math.floor(mxA - margin) - 2);
  const ry = Math.max(0, Math.floor(myA - lip) - 8);
  const w = Math.min(W, Math.ceil(mxA + margin) + 2) - rx;
  const h = Math.min(H, Math.ceil(myA + jaw) + 6) - ry;

  return {
    rx,
    ry,
    w,
    h,
    orig: new Uint8ClampedArray(getRegion(rx, ry, w, h)),
    mx: mxA - rx,
    my: myA - ry,
    hw0,
    jaw,
    lip,
    maxOpen: spec.maxOpen * H,
    margin,
    s: hw0 / REF_HALF_W,
  };
}

/* ------------------------------------------------------------------ */
/* Δόντια                                                               */
/* ------------------------------------------------------------------ */

const REL = [0.8, 0.9, 1, 1, 0.9, 0.8];
const REL_SUM = REL.reduce((a, b) => a + b, 0);
const BOUND: number[] = [0];
for (const r of REL) BOUND.push(BOUND[BOUND.length - 1] + r / REL_SUM);

type Tooth = { i: number; u: number; wpx: number; arch: number };

function toothAt(tx: number, totalW: number): Tooth | null {
  if (tx < 0 || tx >= 1) return null;
  let i = 0;
  while (i < REL.length - 1 && tx >= BOUND[i + 1]) i++;
  const span = BOUND[i + 1] - BOUND[i];
  return { i, u: (tx - BOUND[i]) / span, wpx: totalW * span, arch: Math.abs(i - 2.5) / 2.5 };
}

/** Κάλυψη (0..1) δοντιού στη θέση: ο μυς επάνω-κάτω «κρεμιέται» από τη βάση προς την άκρη */
function toothCoverage(t: Tooth, hh: number, dist: number): number {
  // dist = απόσταση από τη βάση του δοντιού προς την άκρη (pixels), hh = ύψος δοντιού
  const r = Math.min(t.wpx * 0.3, hh * 0.5);
  const a = Math.max(1e-3, r / t.wpx);
  const v = Math.abs(t.u - 0.5);
  const v0 = 0.5 - a;
  const dy = v > v0 ? r * (1 - Math.sqrt(Math.max(0, 1 - ((v - v0) / a) ** 2))) : 0;
  return clamp01(hh - dy - dist + 0.5);
}

/* ------------------------------------------------------------------ */
/* Render                                                               */
/* ------------------------------------------------------------------ */

/**
 * Ζωγραφίζει το στόμα στο `out` (RGBA περιοχής). Γυρίζει false αν το άνοιγμα είναι αμελητέο
 * (τότε ο καλών πρέπει να επαναφέρει την αρχική εικόνα).
 */
export function renderMouth(m: MouthRuntime, open: number, width: number, out: Uint8ClampedArray): boolean {
  const D = open * m.maxOpen;
  if (D < 0.4) return false;

  const { w, h, orig, mx, my, hw0, jaw, lip, margin, s } = m;
  const Ul = D * 0.28; // πόσο ανεβαίνει το πάνω χείλος
  const Dl = D - Ul; // πόσο κατεβαίνει το κάτω χείλος + σαγόνι
  const geoW = Math.min(width, 1.08); // πιο πλατύ από τα χείλη δεν ανοίγει φυσικά
  const hw = hw0 * geoW;
  const c = clamp((1 - geoW) * 0.3, -0.04, 0.16); // «σούφρωμα» για Ο/ΟΥ

  const gX = (x: number) => {
    const t = Math.abs(x - mx) / margin;
    return t >= 1 ? 0 : Math.cos((t * Math.PI) / 2) ** 2;
  };
  // το οριζόντιο σούφρωμα σβήνει κατακόρυφα προς τα όρια της περιοχής (χωρίς ραφή)
  const vTop = Math.max(1, my - lip * 0.8);
  const vBot0 = my + jaw * 0.4;
  const vBot1 = Math.max(vBot0 + 1, h - 1);
  const vw = (y: number) => sstep(y / vTop) * (1 - sstep((y - vBot0) / (vBot1 - vBot0)));
  const shiftAt = (x: number, y: number) => -(x - mx) * c * gX(x) * vw(y);
  const wX = (x: number) => {
    const t = Math.abs(x - mx) / hw;
    return t >= 1 ? 0 : Math.cos((t * Math.PI) / 2) ** 0.8;
  };
  const upW = (ys: number) => (1 - clamp01((my - ys) / lip)) ** 2;
  const dnW = (ys: number) => (1 - clamp01((ys - my) / jaw)) ** 2;

  const maxX = w - 1.001;
  const maxY = h - 1.001;
  // bilinear δείγμα από την αρχική περιοχή
  const sample = (sx: number, sy: number, dst: number[]) => {
    const x = clamp(sx - 0.5, 0, maxX);
    const y = clamp(sy - 0.5, 0, maxY);
    const x0 = Math.floor(x);
    const y0 = Math.floor(y);
    const fx = x - x0;
    const fy = y - y0;
    const i00 = (y0 * w + x0) * 4;
    const i10 = i00 + 4;
    const i01 = i00 + w * 4;
    const i11 = i01 + 4;
    for (let k = 0; k < 3; k++) {
      dst[k] =
        (orig[i00 + k] * (1 - fx) + orig[i10 + k] * fx) * (1 - fy) +
        (orig[i01 + k] * (1 - fx) + orig[i11 + k] * fx) * fy;
    }
  };

  // Εσωτερικό: παράμετροι
  const teethVis = clamp((width - 0.72) / 0.25, 0, 1); // κρύβονται στα Ο/ΟΥ
  const showUpper = open > 0.18 && teethVis > 0;
  const showLower = open > 0.55 && teethVis > 0;
  const lowerVis = teethVis * clamp((open - 0.55) / 0.25, 0, 1);
  const totalW = hw * 1.5;
  const upperCap = 18 * s;
  const lowerCap = 11 * s;
  const showTongue = open > 0.4;

  const interior = (dst: number[], xd: number, yd: number, yTop: number, yBot: number, oh: number, q: number) => {
    const pv = clamp01((yd - yTop) / oh); // 0 πάνω, 1 κάτω
    // κοιλότητα: σκούρο, ζεστό, πιο βαθύ στο κέντρο
    let cr = mix(14, 40, pv);
    let cg = mix(7, 19, pv);
    let cb = mix(9, 23, pv);
    const throat = Math.exp(-(q * q) / 0.32);
    const deep = 1 - 0.6 * throat * (1 - 0.35 * pv);
    cr *= deep;
    cg *= deep;
    cb *= deep;

    // γλώσσα
    if (showTongue) {
      const rxT = hw * 0.46;
      const ryT = Math.max(0.8, Dl * 0.34);
      const cyT = yBot - ryT * 0.55;
      const dx = (xd - mx) / rxT;
      const dy = (yd - cyT) / ryT;
      const nd = Math.sqrt(dx * dx + dy * dy) - 1;
      const cov = clamp01(0.5 - nd * Math.min(rxT, ryT));
      if (cov > 0) {
        const v = clamp01((yd - (cyT - ryT)) / (2 * ryT));
        let tr = mix(118, 78, sstep(v / 0.6));
        let tg = mix(70, 44, sstep(v / 0.6));
        let tb = mix(74, 49, sstep(v / 0.6));
        tr = mix(tr, 44, sstep((v - 0.6) / 0.4));
        tg = mix(tg, 22, sstep((v - 0.6) / 0.4));
        tb = mix(tb, 26, sstep((v - 0.6) / 0.4));
        const groove = 1 - 0.3 * Math.exp(-((xd - mx) * (xd - mx)) / (2 * 1.4 * 1.4)) * (1 - v);
        tr *= groove;
        tg *= groove;
        tb *= groove;
        const hl =
          0.3 * Math.exp(-(((xd - (mx - rxT * 0.2)) / (rxT * 0.3)) ** 2 + ((yd - (cyT - ryT * 0.4)) / (ryT * 0.22)) ** 2));
        tr += (255 - tr) * hl;
        tg += (255 - tg) * hl;
        tb += (255 - tb) * hl;
        cr = mix(cr, tr, cov);
        cg = mix(cg, tg, cov);
        cb = mix(cb, tb, cov);
      }
    }

    // δόντια (κάτω πρώτα, ώστε τα πάνω να είναι μπροστά)
    if (showLower) {
      const tw = totalW * 0.9;
      const tooth = toothAt((xd - mx) / tw + 0.5, tw);
      if (tooth) {
        const hh = Math.max(0.6, Math.min(Dl * 0.4, lowerCap) * (1 - 0.3 * tooth.arch * tooth.arch));
        const cov = toothCoverage(tooth, hh, yBot - yd) * lowerVis;
        if (cov > 0 && yBot - yd >= -0.5) {
          const t = clamp01((yBot - yd) / hh);
          const base = t < 0.3 ? mix(120, 176, t / 0.3) : mix(176, 160, (t - 0.3) / 0.7);
          const edge = 1 - 0.34 * sstep((Math.abs(tooth.u - 0.5) * 2 - 0.45) / 0.55);
          const gap = 1 - 0.5 * (1 - sstep((Math.min(tooth.u, 1 - tooth.u) * tooth.wpx) / 0.9));
          const k2 = (edge * gap * (1 - 0.4 * tooth.arch * tooth.arch)) * 0.88;
          cr = mix(cr, base * k2, cov);
          cg = mix(cg, (base - 4) * k2, cov);
          cb = mix(cb, (base - 14) * k2, cov);
        }
      }
    }
    if (showUpper) {
      const tooth = toothAt((xd - mx) / totalW + 0.5, totalW);
      if (tooth) {
        const hh = Math.max(0.6, Math.min(oh * 0.55, upperCap) * (1 - 0.3 * tooth.arch * tooth.arch));
        const cov = toothCoverage(tooth, hh, yd - yTop) * teethVis;
        if (cov > 0) {
          const t = clamp01((yd - yTop) / hh);
          // βάση σκιασμένη (ούλα), φως στη μέση, ζεστή/ημιδιαφανής άκρη
          let tr = t < 0.35 ? mix(150, 238, t / 0.35) : mix(238, 214, (t - 0.35) / 0.65);
          let tg = t < 0.35 ? mix(142, 234, t / 0.35) : mix(234, 208, (t - 0.35) / 0.65);
          let tb = t < 0.35 ? mix(130, 224, t / 0.35) : mix(224, 192, (t - 0.35) / 0.65);
          const edge = 1 - 0.34 * sstep((Math.abs(tooth.u - 0.5) * 2 - 0.45) / 0.55);
          const gap = 1 - 0.5 * (1 - sstep((Math.min(tooth.u, 1 - tooth.u) * tooth.wpx) / 0.9));
          const dark = edge * gap * (1 - 0.4 * tooth.arch * tooth.arch);
          tr *= dark;
          tg *= dark;
          tb *= dark;
          // γυαλάδα
          const spec = 0.3 * Math.exp(-(((tooth.u - 0.34) / 0.14) ** 2 + ((t - 0.38) / 0.2) ** 2));
          tr += (255 - tr) * spec;
          tg += (255 - tg) * spec;
          tb += (255 - tb) * spec;
          // διαφάνεια του σμάλτου κοντά στην άκρη
          const tl = sstep((t - 0.72) / 0.28) * 0.28;
          tr = mix(tr, cr, tl);
          tg = mix(tg, cg, tl);
          tb = mix(tb, cb, tl);
          cr = mix(cr, tr, cov);
          cg = mix(cg, tg, cov);
          cb = mix(cb, tb, cov);
        }
      }
    }

    // σκιές: από το πάνω χείλος, κάτω χείλος και γωνίες του στόματος
    const ts = Math.exp(-(yd - yTop) / Math.max(2, 0.45 * oh)) * 0.55;
    const bs = Math.exp(-(yBot - yd) / Math.max(1.5, 0.2 * oh)) * 0.3;
    const cs = sstep((Math.abs(q) - 0.6) / 0.4) * 0.75;
    const shade = (1 - ts) * (1 - bs) * (1 - cs);
    dst[0] = cr * shade;
    dst[1] = cg * shade;
    dst[2] = cb * shade;
  };

  const lip3 = [0, 0, 0];
  const col = [0, 0, 0];
  out.set(orig);

  for (let X = 0; X < w; X++) {
    const xd = X + 0.5;
    // αντίστροφη οριζόντια μετατόπιση (σούφρωμα)
    let xs = xd;
    for (let it = 0; it < 3; it++) xs = xd - shiftAt(xs, my);
    const k = wX(xs);
    const moved = Math.abs(shiftAt(xs, my));
    if (k < 0.02 && moved < 0.3) continue; // αμετάβλητη στήλη

    const yTop = my - Ul * k;
    const yBot = my + Dl * k;
    const hasOpen = k >= 0.02 && D * k >= 0.3;
    const oh = Math.max(1e-3, yBot - yTop);
    const q = (xd - mx) / Math.max(1, hw);

    for (let y = 0; y < h; y++) {
      const yd = y + 0.5;

      // κάλυψη ανοίγματος (με anti-alias στα όρια)
      let cin = 0;
      if (hasOpen) cin = clamp01(yd - yTop + 0.5) * clamp01(yBot - yd + 0.5);

      // χείλη: αντίστροφη κατακόρυφη μετατόπιση
      if (cin < 1) {
        let ys: number;
        if (!hasOpen) {
          ys = yd;
        } else if (yd < (yTop + yBot) / 2) {
          const t = Math.min(yd, yTop);
          ys = t;
          for (let it = 0; it < 8; it++) ys = t + Ul * k * upW(ys);
        } else {
          const t = Math.max(yd, yBot);
          ys = t;
          for (let it = 0; it < 8; it++) ys = t - Dl * k * dnW(ys);
        }
        let xsy = xd;
        for (let it = 0; it < 2; it++) xsy = xd - shiftAt(xsy, yd);
        sample(xsy, ys, lip3);
        if (hasOpen) {
          // μικρή σκιά επαφής στα χείλη δίπλα στο άνοιγμα
          const d = yd < (yTop + yBot) / 2 ? yTop - yd : yd - yBot;
          const f = 1 - 0.22 * Math.exp(-Math.max(0, d) / (2.2 * s)) * k;
          lip3[0] *= f;
          lip3[1] *= f;
          lip3[2] *= f;
        }
      }

      let r: number;
      let g: number;
      let b: number;

      if (cin > 0) {
        interior(col, xd, yd, yTop, yBot, oh, q);
        r = lip3[0] * (1 - cin) + col[0] * cin;
        g = lip3[1] * (1 - cin) + col[1] * cin;
        b = lip3[2] * (1 - cin) + col[2] * cin;
        if (cin >= 1) {
          r = col[0];
          g = col[1];
          b = col[2];
        }
      } else {
        r = lip3[0];
        g = lip3[1];
        b = lip3[2];
      }

      const o = (y * w + X) * 4;
      out[o] = r;
      out[o + 1] = g;
      out[o + 2] = b;
      out[o + 3] = 255;

    }
  }
  return true;
}

import { useEffect, useRef, useState } from "react";
import { supabase } from "@/integrations/supabase/client"; // ← προσαρμόσε στο δικό σου path
import { speakable, type AuraCard } from "@/components/aura/AuraChatCards";
import { useAuraSpeech } from "@/hooks/useAuraSpeech";

export type Msg = { role: "user" | "assistant"; content: string; cards?: AuraCard[] };

export const DEFAULT_LIMIT = 10; // μέχρι να απαντήσει ο server με το πραγματικό όριο
const STORE_KEY = "aura_chat_v2";

type Saved = { messages: Msg[]; remaining: number | null; limit: number; limitReached: boolean };

function loadSaved(): Saved | null {
  try {
    const raw = sessionStorage.getItem(STORE_KEY);
    if (!raw) return null;
    const s = JSON.parse(raw) as Saved;
    if (!Array.isArray(s.messages)) return null;
    return {
      messages: s.messages.slice(-20),
      remaining: typeof s.remaining === "number" ? s.remaining : null,
      limit: typeof s.limit === "number" ? s.limit : DEFAULT_LIMIT,
      limitReached: !!s.limitReached,
    };
  } catch {
    return null;
  }
}

function getDeviceId() {
  try {
    let id = localStorage.getItem("aura_device");
    if (!id) {
      id = crypto.randomUUID();
      localStorage.setItem("aura_device", id);
    }
    return id;
  } catch {
    return "unknown";
  }
}

/**
 * Όλη η λογική του AURA AI σε ένα σημείο: η σελίδα /ai και το floating widget μοιράζονται
 * την ίδια συζήτηση, το ίδιο όριο μηνυμάτων, τις ίδιες κάρτες και την ίδια φωνή.
 *
 * `canSpeak`: το widget μιλά μόνο όταν είναι ανοιχτό.
 */
export function useAuraChat(opts: { canSpeak?: () => boolean } = {}) {
  const [messages, setMessages] = useState<Msg[]>([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const [remaining, setRemaining] = useState<number | null>(null);
  const [limit, setLimit] = useState(DEFAULT_LIMIT);
  const [limitReached, setLimitReached] = useState(false);
  const [voiceOn, setVoiceOn] = useState(true);
  const [ready, setReady] = useState(false);

  const canSpeakRef = useRef(opts.canSpeak);
  canSpeakRef.current = opts.canSpeak;

  const { speak, stop, speaking, voiceIssue, getViseme } = useAuraSpeech();

  // φόρτωση της αποθηκευμένης συζήτησης (μόνο στον browser, ώστε να μην χαλάει το SSR)
  useEffect(() => {
    const s = loadSaved();
    if (s) {
      setMessages(s.messages);
      setRemaining(s.remaining);
      setLimit(s.limit);
      setLimitReached(s.limitReached);
    }
    setReady(true);
  }, []);

  useEffect(() => {
    if (!ready) return;
    try {
      sessionStorage.setItem(
        STORE_KEY,
        JSON.stringify({ messages: messages.slice(-20), remaining, limit, limitReached } satisfies Saved),
      );
    } catch {
      /* ignore */
    }
  }, [ready, messages, remaining, limit, limitReached]);

  async function send(override?: string) {
    const text = (override ?? input).trim();
    if (!text || loading || limitReached) return;

    stop();
    const next: Msg[] = [...messages, { role: "user", content: text }];
    setMessages(next);
    setInput("");
    setLoading(true);

    try {
      let reply: string | null = null;
      let cards: AuraCard[] | undefined;
      const payload = next.map(({ role, content }) => ({ role, content })); // στον server μόνο κείμενο

      for (let attempt = 0; attempt < 2 && !reply; attempt++) {
        try {
          const { data, error: invokeErr } = await supabase.functions.invoke("ai-chat", {
            body: { messages: payload, deviceId: getDeviceId() },
          });
          if (invokeErr) throw invokeErr;

          if (data?.error === "limit_reached") {
            setLimitReached(true);
            setRemaining(0);
            return;
          }
          if (typeof data?.reply === "string" && data.reply.trim()) {
            reply = data.reply;
            cards = Array.isArray(data.cards) && data.cards.length ? (data.cards as AuraCard[]) : undefined;
            if (typeof data.remaining === "number") setRemaining(data.remaining);
            if (typeof data.limit === "number") setLimit(data.limit);
          }
        } catch (e) {
          console.error(e);
        }
      }

      const finalReply = reply ?? "Συγγνώμη, είχα ένα μικρό πρόβλημα σύνδεσης. Στείλε μου ξανά το μήνυμά σου.";
      setMessages([...next, { role: "assistant", content: finalReply, cards }]);
      // η φωνή διαβάζει μόνο τις πρώτες προτάσεις (λιγότερη κατανάλωση TTS)
      if (voiceOn && (canSpeakRef.current ? canSpeakRef.current() : true)) void speak(speakable(finalReply));
    } finally {
      setLoading(false);
    }
  }

  function newChat() {
    stop();
    setMessages([]);
  }

  function toggleVoice() {
    if (voiceOn) stop();
    setVoiceOn(!voiceOn);
  }

  function replay(text: string) {
    void speak(speakable(text));
  }

  return {
    ready,
    messages,
    input,
    setInput,
    loading,
    remaining,
    limit,
    limitReached,
    left: remaining ?? limit,
    voiceOn,
    toggleVoice,
    speaking,
    voiceIssue,
    getViseme,
    stopVoice: stop,
    send,
    newChat,
    replay,
  };
}

export type AuraChat = ReturnType<typeof useAuraChat>;

import { useEffect, useState } from "react";

export function useMediaQuery(query: string): boolean {
  const [match, setMatch] = useState(false);
  useEffect(() => {
    const mq = window.matchMedia(query);
    const update = () => setMatch(mq.matches);
    update();
    mq.addEventListener("change", update);
    return () => mq.removeEventListener("change", update);
  }, [query]);
  return match;
}

/**
 * Στο κινητό η σελίδα «κλειδώνει» στο ορατό τμήμα της οθόνης (visualViewport). Όταν ανοίγει το
 * πληκτρολόγιο, όλο το UI μικραίνει ώστε να χωράει πάνω από αυτό, και ο χαρακτήρας μένει ορατός.
 * Η σελίδα πίσω δεν κινείται. Όταν enabled γίνει false, όλα επανέρχονται όπως ήταν.
 */
export function useLockedViewport(enabled: boolean) {
  const [vp, setVp] = useState({ height: 0, top: 0, keyboard: false });

  useEffect(() => {
    if (!enabled) return;
    const vv = window.visualViewport;
    let maxH = 0;
    let lastW = window.innerWidth;

    const update = () => {
      if (window.innerWidth !== lastW) {
        lastW = window.innerWidth; // περιστροφή οθόνης
        maxH = 0;
      }
      const height = Math.round(vv ? vv.height : window.innerHeight);
      const top = Math.round(vv ? vv.offsetTop : 0);
      maxH = Math.max(maxH, height); // το μεγαλύτερο ύψος που είδαμε = πληκτρολόγιο κλειστό
      const keyboard = maxH - height > 120;
      setVp((p) => (p.height === height && p.top === top && p.keyboard === keyboard ? p : { height, top, keyboard }));
      if (window.scrollY !== 0) window.scrollTo(0, 0); // το iOS σκρολάρει τη σελίδα όταν εστιάζεις input
    };
    update();

    vv?.addEventListener("resize", update);
    vv?.addEventListener("scroll", update);
    window.addEventListener("resize", update);
    window.addEventListener("orientationchange", update);

    // κλείδωσε τη σελίδα πίσω από το chat
    const html = document.documentElement;
    const body = document.body;
    const scrollY = window.scrollY;
    const prev = {
      hOverflow: html.style.overflow,
      hOverscroll: html.style.overscrollBehavior,
      bOverflow: body.style.overflow,
      bPosition: body.style.position,
      bTop: body.style.top,
      bWidth: body.style.width,
    };
    html.style.overflow = "hidden";
    html.style.overscrollBehavior = "none";
    body.style.overflow = "hidden";
    body.style.position = "fixed";
    body.style.top = `-${scrollY}px`;
    body.style.width = "100%";

    return () => {
      vv?.removeEventListener("resize", update);
      vv?.removeEventListener("scroll", update);
      window.removeEventListener("resize", update);
      window.removeEventListener("orientationchange", update);
      html.style.overflow = prev.hOverflow;
      html.style.overscrollBehavior = prev.hOverscroll;
      body.style.overflow = prev.bOverflow;
      body.style.position = prev.bPosition;
      body.style.top = prev.bTop;
      body.style.width = prev.bWidth;
      window.scrollTo(0, scrollY);
    };
  }, [enabled]);

  return vp;
}
