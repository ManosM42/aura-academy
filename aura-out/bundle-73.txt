  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
  type ReactNode,

import { supabase } from "@/lib/supabase";
import { useAuth } from "@/lib/useAuth";
import { formatDuration, type CallLog } from "@/lib/call-logs";

type Kind = "audio" | "video";
type Phase = "idle" | "ringing" | "connecting" | "connected";

export interface CallPeer {
  id: string;
  name: string;
  avatar: string | null;
}
interface CallInfo {
  id: string;
  conversationId: string;
  kind: Kind;
  peer: CallPeer;
  direction: "out" | "in";
}
interface CallCtx {
  busy: boolean;
  startCall: (peer: CallPeer, conversationId: string, kind: Kind) => Promise<void>;
}

const Ctx = createContext<CallCtx>({ busy: false, startCall: async () => {} });
export const useCall = () => useContext(Ctx);

const RING_TIMEOUT_MS = 45_000;
const FALLBACK_ICE: RTCIceServer[] = [{ urls: "stun:stun.l.google.com:19302" }];

function makeId(): string {
  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
    return crypto.randomUUID();
  }
  const b = new Uint8Array(16);
  crypto.getRandomValues(b);
  b[6] = (b[6] & 0x0f) | 0x40;
  b[8] = (b[8] & 0x3f) | 0x80;
  const h = [...b].map((x) => x.toString(16).padStart(2, "0")).join("");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20)}`;
}

async function getIceServers(): Promise<RTCIceServer[]> {
  try {
    const { data, error } = await supabase.functions.invoke("get-ice-servers");
    if (!error && data?.iceServers) return data.iceServers as RTCIceServer[];
  } catch {
    /* fall through */
  }
  return FALLBACK_ICE;
}

function subscribe(ch: RealtimeChannel, timeoutMs = 8000): Promise<void> {
  return new Promise((resolve, reject) => {
    const t = setTimeout(() => reject(new Error("subscribe timeout")), timeoutMs);
    ch.subscribe((status) => {
      if (status === "SUBSCRIBED") {
        clearTimeout(t);
        resolve();
      }
      if (status === "CHANNEL_ERROR" || status === "TIMED_OUT" || status === "CLOSED") {
        clearTimeout(t);
        reject(new Error(status));
      }
    });
  });
}

/** Tries a private channel first, then falls back to a public one. */
async function openChannel(
  name: string,
  setup: (ch: RealtimeChannel) => void,
): Promise<RealtimeChannel> {
  let lastErr: unknown;
  for (const priv of [true, false]) {
    const ch = supabase.channel(name, { config: { private: priv } });
    setup(ch);
    try {
      await subscribe(ch);
      return ch;
    } catch (e) {
      lastErr = e;
      supabase.removeChannel(ch);
    }
  }
  throw lastErr;
}

export function CallProvider({ children }: { children: ReactNode }) {
  const { profile } = useAuth();
  const myId = profile?.id;

  const [call, setCall] = useState<CallInfo | null>(null);
  const [phase, setPhase] = useState<Phase>("idle");
  const [muted, setMuted] = useState(false);
  const [camOff, setCamOff] = useState(false);
  const [localStream, setLocalStream] = useState<MediaStream | null>(null);
  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null);
  const [seconds, setSeconds] = useState(0);
  const [notice, setNotice] = useState<string | null>(null);

  const callRef = useRef<CallInfo | null>(null);
  const phaseRef = useRef<Phase>("idle");
  const pcRef = useRef<RTCPeerConnection | null>(null);
  const roomRef = useRef<RealtimeChannel | null>(null);
  const localRef = useRef<MediaStream | null>(null);
  const pendingIce = useRef<RTCIceCandidateInit[]>([]);
  const ringTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const noticeTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const readyTimer = useRef<ReturnType<typeof setInterval> | null>(null);
  const connTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const offeringRef = useRef(false);

  const localVideoRef = useRef<HTMLVideoElement>(null);
  const remoteVideoRef = useRef<HTMLVideoElement>(null);

  const setPhaseBoth = (p: Phase) => {
    phaseRef.current = p;
    setPhase(p);
  };

  const showNotice = useCallback((msg: string) => {
    setNotice(msg);
    if (noticeTimer.current) clearTimeout(noticeTimer.current);
    noticeTimer.current = setTimeout(() => setNotice(null), 6000);
  }, []);

  const cleanup = useCallback(() => {
    if (ringTimer.current) clearTimeout(ringTimer.current);
    ringTimer.current = null;
    if (readyTimer.current) clearInterval(readyTimer.current);
    readyTimer.current = null;
    if (connTimer.current) clearTimeout(connTimer.current);
    connTimer.current = null;
    offeringRef.current = false;
    localRef.current?.getTracks().forEach((t) => t.stop());
    localRef.current = null;
    pcRef.current?.close();
    pcRef.current = null;
    if (roomRef.current) supabase.removeChannel(roomRef.current);
    roomRef.current = null;
    pendingIce.current = [];
    callRef.current = null;
    setCall(null);
    setLocalStream(null);
    setRemoteStream(null);
    setMuted(false);
    setCamOff(false);
    setSeconds(0);
    setPhaseBoth("idle");
  }, []);

  useEffect(() => {
    if (localVideoRef.current) localVideoRef.current.srcObject = localStream;
  }, [localStream, phase]);
  useEffect(() => {
    if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remoteStream;
  }, [remoteStream, phase]);

  useEffect(() => {
    if (phase !== "connected") return;
    const t = setInterval(() => setSeconds((s) => s + 1), 1000);
    return () => clearInterval(t);
  }, [phase]);

  // simple ringtone (incoming = higher beep, outgoing = lower beep)
  const ringing = !!call && phase === "ringing";
  const incomingDir = call?.direction === "in";
  useEffect(() => {
    if (!ringing) return;
    let ctx: AudioContext | null = null;
    try {
      const AC = window.AudioContext || (window as any).webkitAudioContext;
      ctx = new AC();
    } catch {
      return;
    }
    const beep = () => {
      if (!ctx) return;
      try {
        const o = ctx.createOscillator();
        const g = ctx.createGain();
        o.frequency.value = incomingDir ? 880 : 440;
        g.gain.value = 0.05;
        o.connect(g);
        g.connect(ctx.destination);
        o.start();
        o.stop(ctx.currentTime + 0.4);
      } catch {
        /* ignore */
      }
    };
    beep();
    const t = setInterval(() => {
      beep();
      if (incomingDir && navigator.vibrate) navigator.vibrate([300, 150, 300]);
    }, 1500);
    return () => {
      clearInterval(t);
      ctx?.close().catch(() => {});
    };
  }, [ringing, incomingDir]);

  // ---------- end / hang up ----------
  const endCall = useCallback(
    async (finalStatus: "ended" | "missed" = "ended") => {
      const info = callRef.current;
      if (!info) return;
      const answered = phaseRef.current === "connected" || phaseRef.current === "connecting";
      const status = answered
        ? "ended"
        : info.direction === "out"
          ? finalStatus === "missed"
            ? "missed"
            : "cancelled"
          : "ended";
      await supabase
        .from("call_logs")
        .update({ status, ended_at: new Date().toISOString() })
        .eq("id", info.id)
        .is("ended_at", null);
      roomRef.current?.send({ type: "broadcast", event: "hangup", payload: {} });
      setTimeout(cleanup, 80);
    },
    [cleanup],
  );

  // ---------- peer connection + signaling room ----------
  const joinRoom = useCallback(
    async (info: CallInfo, isCaller: boolean) => {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: true,
        video: info.kind === "video" ? { facingMode: "user" } : false,
      });
      localRef.current = stream;
      setLocalStream(stream);

      const pc = new RTCPeerConnection({ iceServers: await getIceServers() });
      pcRef.current = pc;
      stream.getTracks().forEach((t) => pc.addTrack(t, stream));

      const remote = new MediaStream();
      pc.ontrack = (e) => {
        e.streams[0]?.getTracks().forEach((t) => {
          if (!remote.getTracks().includes(t)) remote.addTrack(t);
        });
        setRemoteStream(new MediaStream(remote.getTracks()));
      };
      const markConnected = () => {
        if (connTimer.current) clearTimeout(connTimer.current);
        connTimer.current = null;
        setPhaseBoth("connected");
      };
      const startConnTimer = () => {
        if (connTimer.current) return;
        connTimer.current = setTimeout(() => {
          if (phaseRef.current !== "connected") {
            showNotice("Δεν έγινε σύνδεση (δίκτυο/TURN). Δοκίμασε ξανά.");
            void endCall("ended");
          }
        }, 25_000);
      };
      pc.oniceconnectionstatechange = () => {
        console.log("[call] ice:", pc.iceConnectionState);
        if (pc.iceConnectionState === "connected" || pc.iceConnectionState === "completed") {
          markConnected();
        }
      };
      pc.onconnectionstatechange = () => {
        console.log("[call] pc:", pc.connectionState);
        if (pc.connectionState === "connected") markConnected();
        if (pc.connectionState === "failed") {
          showNotice("Η σύνδεση απέτυχε.");
          void endCall("ended");
        }
      };

      const send = (event: string, payload: unknown = {}) =>
        roomRef.current?.send({ type: "broadcast", event, payload });

      pc.onicecandidate = (e) => {
        if (e.candidate) void send("ice", { c: e.candidate.toJSON() });
      };

      const flushIce = async () => {
        for (const c of pendingIce.current) await pc.addIceCandidate(c).catch(() => {});
        pendingIce.current = [];
      };

      roomRef.current = await openChannel(`call:${info.id}`, (room) => {
        room
          .on("broadcast", { event: "ready" }, async () => {
            if (!isCaller) return;
            console.log("[call] got ready");
            if (ringTimer.current) clearTimeout(ringTimer.current);
            ringTimer.current = null;
            if (phaseRef.current === "ringing") setPhaseBoth("connecting");
            startConnTimer();
            if (pc.remoteDescription) return; // already answered
            if (pc.localDescription) {
              void send("offer", { sdp: pc.localDescription }); // callee asked again
              return;
            }
            if (offeringRef.current) return;
            offeringRef.current = true;
            const offer = await pc.createOffer();
            await pc.setLocalDescription(offer);
            void send("offer", { sdp: offer });
          })
          .on("broadcast", { event: "offer" }, async ({ payload }) => {
            if (pc.remoteDescription) return; // duplicate offer
            console.log("[call] got offer");
            if (readyTimer.current) clearInterval(readyTimer.current);
            readyTimer.current = null;
            await pc.setRemoteDescription(payload.sdp);
            await flushIce();
            const answer = await pc.createAnswer();
            await pc.setLocalDescription(answer);
            void send("answer", { sdp: answer });
          })
          .on("broadcast", { event: "answer" }, async ({ payload }) => {
            if (pc.remoteDescription) return;
            console.log("[call] got answer");
            await pc.setRemoteDescription(payload.sdp);
            await flushIce();
          })
          .on("broadcast", { event: "ice" }, async ({ payload }) => {
            if (pc.remoteDescription) await pc.addIceCandidate(payload.c).catch(() => {});
            else pendingIce.current.push(payload.c);
          })
          .on("broadcast", { event: "hangup" }, () => cleanup())
          .on("broadcast", { event: "decline" }, () => cleanup());
      });

      return send;
    },
    [cleanup, endCall, showNotice],
  );

  const mediaErrorText = (e: unknown) => {
    const name = (e as Error)?.name;
    if (name === "NotAllowedError") return "Επίτρεψε μικρόφωνο/κάμερα στον browser.";
    if (name === "NotFoundError") return "Δεν βρέθηκε μικρόφωνο/κάμερα.";
    if (!window.isSecureContext) return "Οι κλήσεις θέλουν HTTPS.";
    return "Η κλήση απέτυχε. Δοκίμασε ξανά.";
  };

  // ---------- outgoing ----------
  const startCall = useCallback(
    async (peer: CallPeer, conversationId: string, kind: Kind) => {
      if (!profile || callRef.current) return;
      const id = makeId();
      const info: CallInfo = { id, conversationId, kind, peer, direction: "out" };
      callRef.current = info;
      setCall(info);
      setPhaseBoth("ringing");

      try {
        // 1) the INSERT makes the other phone ring (and lets us join the private room)
        const { error } = await supabase.from("call_logs").insert({
          id,
          conversation_id: conversationId,
          caller_id: profile.id,
          callee_id: peer.id,
          kind,
        });
        if (error) throw error;

        // 2) camera/mic + signaling room (callee keeps asking "ready" until we answer)
        await joinRoom(info, true);

        ringTimer.current = setTimeout(() => {
          if (phaseRef.current === "ringing") void endCall("missed");
        }, RING_TIMEOUT_MS);
      } catch (e) {
        console.error("startCall failed", e);
        showNotice(mediaErrorText(e));
        await supabase
          .from("call_logs")
          .update({ status: "cancelled", ended_at: new Date().toISOString() })
          .eq("id", id);
        cleanup();
      }
    },
    [profile, joinRoom, endCall, cleanup, showNotice],
  );

  // ---------- incoming: listen for new call_logs rows addressed to me ----------
  useEffect(() => {
    if (!myId) return;

    const showIncoming = async (row: CallLog) => {
      if (callRef.current) return; // already busy or already showing this one
      if (row.status !== "ringing" || row.ended_at) return;
      if (Date.now() - new Date(row.created_at).getTime() > 60_000) return;

      let name = "Χρήστης";
      let avatar: string | null = null;
      try {
        const { data } = await supabase
          .from("profiles")
          .select("full_name, avatar_url")
          .eq("id", row.caller_id)
          .maybeSingle();
        if (data) {
          name = (data as any).full_name ?? name;
          avatar = (data as any).avatar_url ?? null;
        }
      } catch {
        /* keep defaults */
      }
      if (callRef.current) return;

      const info: CallInfo = {
        id: row.id,
        conversationId: row.conversation_id,
        kind: row.kind,
        peer: { id: row.caller_id, name, avatar },
        direction: "in",
      };
      callRef.current = info;
      setCall(info);
      setPhaseBoth("ringing");
      if (navigator.vibrate) navigator.vibrate([300, 150, 300]);
    };

    const ch = supabase
      .channel(`incoming-calls:${myId}`)
      .on(
        "postgres_changes",
        { event: "INSERT", schema: "public", table: "call_logs", filter: `callee_id=eq.${myId}` },
        (payload) => void showIncoming(payload.new as CallLog),
      )
      .subscribe();

    // catch a call that started while this page was loading
    (async () => {
      const since = new Date(Date.now() - 45_000).toISOString();
      const { data } = await supabase
        .from("call_logs")
        .select("*")
        .eq("callee_id", myId)
        .eq("status", "ringing")
        .is("ended_at", null)
        .gte("created_at", since)
        .order("created_at", { ascending: false })
        .limit(1);
      if (data?.[0]) void showIncoming(data[0] as CallLog);
    })();

    return () => {
      supabase.removeChannel(ch);
    };
  }, [myId]);

  // ---------- watch the current call's row: ended / declined / cancelled ----------
  const callId = call?.id;
  useEffect(() => {
    if (!callId) return;
    const ch = supabase
      .channel(`call-status:${callId}`)
      .on(
        "postgres_changes",
        { event: "UPDATE", schema: "public", table: "call_logs", filter: `id=eq.${callId}` },
        (payload) => {
          const row = payload.new as CallLog;
          if (row.ended_at && callRef.current?.id === callId) cleanup();
        },
      )
      .subscribe();
    return () => {
      supabase.removeChannel(ch);
    };
  }, [callId, cleanup]);

  const accept = async () => {
    const info = callRef.current;
    if (!info) return;
    setPhaseBoth("connecting");
    try {
      const send = await joinRoom(info, false);
      await supabase
        .from("call_logs")
        .update({ status: "answered", answered_at: new Date().toISOString() })
        .eq("id", info.id);
      let tries = 0;
      const sendReady = () => {
        console.log("[call] send ready");
        void send("ready");
      };
      sendReady();
      readyTimer.current = setInterval(() => {
        if (pcRef.current?.remoteDescription || ++tries > 25) {
          if (readyTimer.current) clearInterval(readyTimer.current);
          readyTimer.current = null;
          return;
        }
        sendReady();
      }, 1000);
      connTimer.current = setTimeout(() => {
        if (phaseRef.current !== "connected") {
          showNotice("Δεν έγινε σύνδεση (δίκτυο/TURN). Δοκίμασε ξανά.");
          void endCall("ended");
        }
      }, 25_000);
    } catch (e) {
      console.error("accept failed", e);
      showNotice(mediaErrorText(e));
      await endCall("ended");
    }
  };

  const decline = async () => {
    const info = callRef.current;
    if (!info) return;
    await supabase
      .from("call_logs")
      .update({ status: "declined", ended_at: new Date().toISOString() })
      .eq("id", info.id);
    cleanup();
  };

  const toggleMute = () => {
    const next = !muted;
    localRef.current?.getAudioTracks().forEach((t) => (t.enabled = !next));
    setMuted(next);
  };
  const toggleCam = () => {
    const next = !camOff;
    localRef.current?.getVideoTracks().forEach((t) => (t.enabled = !next));
    setCamOff(next);
  };

  const value = useMemo(() => ({ busy: !!call, startCall }), [call, startCall]);
  const incomingRinging = call?.direction === "in" && phase === "ringing";
  const isVideo = call?.kind === "video";

  return (
    <Ctx.Provider value={value}>
      {children}

      {notice && (
        <div className="fixed left-1/2 top-4 z-[120] -translate-x-1/2 rounded-full bg-red-500/90 px-4 py-2 text-sm text-white shadow-lg">
          {notice}
        </div>
      )}

      {call && (
        <div className="fixed inset-0 z-[100] flex flex-col bg-[#070707] text-white">
          <video
            ref={remoteVideoRef}
            autoPlay
            playsInline
            className={`absolute inset-0 h-full w-full object-cover ${
              isVideo && phase === "connected" ? "" : "invisible"
            }`}
          />

          {(!isVideo || phase !== "connected") && (
            <div className="relative z-10 flex flex-1 flex-col items-center justify-center gap-4 px-6">
              {call.peer.avatar ? (
                <img src={call.peer.avatar} alt="" className="h-28 w-28 rounded-full object-cover ring-1 ring-white/20" />
              ) : (
                <div className="flex h-28 w-28 items-center justify-center rounded-full bg-white/10 text-3xl font-semibold">
                  {call.peer.name.charAt(0).toUpperCase()}
                </div>
              )}
              <p className="text-xl font-medium">{call.peer.name}</p>
              <p className="text-sm text-white/50">
                {phase === "connected"
                  ? formatDuration(seconds)
                  : phase === "connecting"
                    ? "Σύνδεση…"
                    : incomingRinging
                      ? isVideo ? "Εισερχόμενη βιντεοκλήση" : "Εισερχόμενη κλήση"
                      : "Καλεί…"}
              </p>

          {isVideo && phase === "connected" && (

// Helper to extract item config from a payload.
function getPayloadConfigFromPayload(config: ChartConfig, payload: unknown, key: string) {
  if (typeof payload !== "object" || payload === null) {
    return undefined;
  }

  const payloadPayload =
    "payload" in payload && typeof payload.payload === "object" && payload.payload !== null
      ? payload.payload
      : undefined;

  let configLabelKey: string = key;

  if (key in payload && typeof payload[key as keyof typeof payload] === "string") {
    configLabelKey = payload[key as keyof typeof payload] as string;
  } else if (
    payloadPayload &&
    key in payloadPayload &&
    typeof payloadPayload[key as keyof typeof payloadPayload] === "string"
  ) {
    configLabelKey = payloadPayload[key as keyof typeof payloadPayload] as string;
  }

  return configLabelKey in config ? config[configLabelKey] : config[key as keyof typeof config];
}

    }
  }

  const enabled = state === "enabled";
  const blocked = state === "unsupported" || state === "insecure" || state === "denied";

  return (
    <motion.div
      initial="hidden"
      animate="show"
      variants={{ show: { transition: { staggerChildren: 0.1 } } }}
      className="space-y-10"
    >
      {/* header */}
      <motion.header
        variants={{ hidden: { opacity: 0, y: 18 }, show: { opacity: 1, y: 0, transition: { duration: 0.7, ease: EASE } } }}
        className="border-b border-white/10 pb-8"
      >
        <Link
          to={"/admin" as never}
          className="mb-5 inline-flex items-center gap-2 text-xs text-white/50 outline-none transition-colors hover:text-white focus-visible:text-white"
        >
          <ArrowLeft className="size-3.5" /> Πίσω στο Admin
        </Link>
        <div className="mb-3 flex items-center gap-3 text-xs uppercase tracking-[0.35em] text-white/50">
          <span className="h-px w-10 bg-gradient-to-r from-white/70 to-transparent" />
          Alerts
        </div>
        <h1 className="text-3xl font-extrabold uppercase tracking-tight sm:text-5xl">
          <ChromeText>Ειδοποιήσεις</ChromeText>
        </h1>
      </motion.header>

      {/* enable card */}
      <motion.section variants={{ hidden: { opacity: 0, y: 18 }, show: { opacity: 1, y: 0, transition: { duration: 0.7, ease: EASE } } }}>
        <Frame inner="p-6 sm:p-8">
          <div className="flex flex-col gap-6 sm:flex-row sm:items-center sm:justify-between">
            <div className="flex items-start gap-5">
              <div className="relative shrink-0">
                {enabled && !reduce && (
                  <motion.span
                    aria-hidden
                    className="absolute inset-0 rounded-2xl border border-white/60"
                    animate={{ scale: [1, 1.5], opacity: [0.6, 0] }}
                    transition={{ duration: 2.2, repeat: Infinity, ease: "easeOut" }}
                  />
                )}
                <span
                  className="relative flex size-14 items-center justify-center rounded-2xl text-neutral-950"
                  style={
                    enabled
                      ? { backgroundImage: CHROME_FILL, boxShadow: "0 0 34px rgba(255,255,255,0.5), inset 0 1px 0 rgba(255,255,255,0.95)" }
                      : { background: "rgba(255,255,255,0.06)", color: "rgba(255,255,255,0.6)", boxShadow: "inset 0 0 0 1px rgba(255,255,255,0.15)" }
                  }
                >
                  {enabled ? <BellRing className="size-6" /> : blocked ? <BellOff className="size-6" /> : <Bell className="size-6" />}
                </span>
              </div>
              <div>
                <h2 className="text-lg font-bold text-white">
                  {state === null ? "Έλεγχος…" : enabled ? "Οι ειδοποιήσεις είναι ενεργές" : "Οι ειδοποιήσεις είναι ανενεργές"}
                </h2>
                <p className="mt-1 max-w-md text-sm leading-relaxed text-white/55">
                  {state === "insecure" &&
                    "Το site τρέχει σε http. Οι ειδοποιήσεις του Chrome λειτουργούν μόνο σε https (ή localhost)."}
                  {state === "unsupported" && "Αυτός ο browser δεν υποστηρίζει push ειδοποιήσεις."}
                  {state === "denied" &&
                    "Έχεις μπλοκάρει τις ειδοποιήσεις. Επίτρεψέ τες από το εικονίδιο δίπλα στη διεύθυνση του site και ανανέωσε τη σελίδα."}
                  {(state === "enabled" || state === "disabled") &&
                    "Ενεργοποίησέ τες σε κάθε browser ή συσκευή όπου θέλεις να λαμβάνεις ειδοποιήσεις. Ο Chrome πρέπει να είναι ανοιχτός."}
                </p>
              </div>
            </div>

            <div className="flex shrink-0 flex-col gap-3 sm:items-end">
              <ChromeCTA type="button" solid={!enabled} onClick={() => void toggle()} disabled={busy || blocked || state === null}>
                {busy ? <Loader2 className="size-4 animate-spin" /> : enabled ? <BellOff className="size-4" /> : <BellRing className="size-4" />}
                {enabled ? "Απενεργοποίηση" : "Ενεργοποίηση"}
              </ChromeCTA>
              {enabled && (
                <ChromeCTA type="button" solid={false} onClick={() => void test()} disabled={busy}>
                  <Send className="size-4" /> Δοκιμαστική
                </ChromeCTA>
              )}
            </div>
          </div>

          <AnimatePresence>
            {testSent && (
              <motion.p
                initial={{ opacity: 0, y: -6 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0 }}
                className="mt-5 text-sm font-semibold text-white drop-shadow-[0_0_12px_rgba(255,255,255,0.6)]"
              >
                Στάλθηκε. Η ειδοποίηση πρέπει να εμφανιστεί σε λίγα δευτερόλεπτα.
              </motion.p>
            )}
          </AnimatePresence>
          {error && (
            <p role="alert" className="mt-5 flex items-start gap-2 rounded-xl border border-white/20 bg-white/[0.05] p-3 text-sm text-white/85">
              <AlertCircle className="mt-0.5 size-4 shrink-0" /> {error}
            </p>
          )}
        </Frame>
      </motion.section>

      {/* what notifies */}
      <motion.section variants={{ hidden: { opacity: 0, y: 18 }, show: { opacity: 1, y: 0, transition: { duration: 0.7, ease: EASE } } }}>
        <SectionTitle>Πότε ειδοποιείσαι</SectionTitle>
        <div className="grid gap-4 sm:grid-cols-2">
          {EVENTS.map((e, i) => (
            <motion.div
              key={e.type}
              whileHover={{ y: -4 }}
              className={`group ${i === EVENTS.length - 1 && EVENTS.length % 2 === 1 ? "sm:col-span-2" : ""}`}
            >
              <Frame inner="p-5">
                <div className="flex items-start gap-4">
                  <span className="flex size-11 shrink-0 items-center justify-center rounded-xl border border-white/15 bg-black/60 text-zinc-200 transition-all duration-300 group-hover:border-white/50 group-hover:text-white group-hover:shadow-[0_0_24px_rgba(255,255,255,0.35)]">
                    <e.icon className="size-5" />
                  </span>
                  <div>
                    <h3 className="text-sm font-semibold text-white">{e.label}</h3>
                    <p className="mt-1 text-sm leading-relaxed text-white/50">{e.text}</p>
                  </div>
                </div>
              </Frame>
            </motion.div>
          ))}
        </div>
      </motion.section>

      {/* history */}
      <motion.section variants={{ hidden: { opacity: 0, y: 18 }, show: { opacity: 1, y: 0, transition: { duration: 0.7, ease: EASE } } }}>
        <SectionTitle>Πρόσφατες ειδοποιήσεις</SectionTitle>
        {listError && <ErrorState message={listError} />}
        {!items && !listError && <LoadingSkeleton rows={3} />}
        {items && items.length === 0 && (
          <Frame inner="p-8 text-center">
            <p className="text-sm text-white/45">Δεν υπάρχουν ειδοποιήσεις ακόμη.</p>
          </Frame>
        )}
        {items && items.length > 0 && (
          <Frame inner="p-2 sm:p-3">
            <ul className="divide-y divide-white/[0.06]">
              <AnimatePresence initial={false}>
                {items.map((n) => {
                  const Icon = iconFor(n.type);
                  const row = (
                    <div className="flex items-start gap-4 rounded-xl px-3 py-3.5 transition-colors hover:bg-white/[0.04]">
                      <span className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-lg border border-white/15 bg-white/[0.05] text-white/80">
                        <Icon className="size-4" />
                      </span>
                      <div className="min-w-0 flex-1">
                        <p className="text-sm font-semibold text-white">{n.title}</p>
                        {n.body && <p className="mt-0.5 break-words text-sm text-white/55">{n.body}</p>}
                      </div>
                      <span className="shrink-0 text-xs text-white/35">{dateFmt.format(new Date(n.created_at))}</span>
                    </div>
                  );
                  return (
                    <motion.li
                      key={n.id}
                      layout
                      initial={{ opacity: 0, y: -12 }}
                      animate={{ opacity: 1, y: 0 }}
                      transition={{ duration: 0.4, ease: EASE }}
                    >
                      {n.url && n.type !== "test" ? <Link to={n.url as never}>{row}</Link> : row}
                    </motion.li>
                  );
                })}
              </AnimatePresence>
            </ul>
          </Frame>
        )}
      </motion.section>
    </motion.div>
  );
}

function SectionTitle({ children }: { children: React.ReactNode }) {
  return (
    <h2 className="mb-4 flex items-center gap-3 px-1 text-xs font-semibold uppercase tracking-[0.3em] text-white/65">
      <span className="size-1.5 rounded-full bg-white shadow-[0_0_10px_rgba(255,255,255,0.9)]" />
      {children}
    </h2>
  );
}
