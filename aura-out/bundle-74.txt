/** The line shown inside the chat for a call. */
function describeCall(log, myId) {
	const kind = log.kind === "video" ? "Βιντεοκλήση" : "Ηχητική κλήση";
	const iCalled = log.caller_id === myId;
	if (log.answered_at) {
		if (!log.ended_at) return {
			label: `${kind} σε εξέλιξη`,
			missed: false
		};
		return {
			label: `${kind} · ${formatDuration((new Date(log.ended_at).getTime() - new Date(log.answered_at).getTime()) / 1e3)}`,
			missed: false
		};
	}
	if (log.status === "declined") return {
		label: iCalled ? "Η κλήση απορρίφθηκε" : "Απέρριψες την κλήση",
		missed: false
	};

		label: `${kind}…`,
		missed: false
	};
	return {
		label: iCalled ? "Δεν απάντησε" : "Αναπάντητη κλήση",
		missed: !iCalled
	};
}

function subscribe(ch, timeoutMs = 8e3) {
	return new Promise((resolve, reject) => {
		const t = setTimeout(() => reject(/* @__PURE__ */ new Error("subscribe timeout")), timeoutMs);
		ch.subscribe((status) => {
			if (status === "SUBSCRIBED") {
				clearTimeout(t);
				resolve();
			}
			if (status === "CHANNEL_ERROR" || status === "TIMED_OUT" || status === "CLOSED") {
				clearTimeout(t);
				reject(new Error(status));
			}
		});
	});
}
/** Tries a private channel first, then falls back to a public one. */
async function openChannel(name, setup) {
	let lastErr;
	for (const priv of [true, false]) {

		try {

			return ch;
		} catch (e) {

		}
	}

}

The uploaded AURA logo is the definitive source of truth for the visual identity.

DO NOT introduce unrelated accent colors.

- metallic silver

- bright white highlights

- dark gunmetal shadows

- reflective surfaces

- high contrast

- sophisticated monochrome palette

The website must feel like the digital extension of this exact logo.

- decorative lines

- selected UI highlights

BLACK SPACE + LIGHT + CHROME.
