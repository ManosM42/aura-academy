/** Notifications need HTTPS (or localhost): plain http sites cannot use them. */
function pushState() {
	if (typeof window === "undefined") return "unsupported";
	if (!window.isSecureContext) return "insecure";
	if (!("serviceWorker" in navigator) || !("PushManager" in window) || !("Notification" in window)) return "unsupported";
	if (Notification.permission === "denied") return "denied";
	return "check";
}
/** Is THIS browser currently subscribed? */
async function getPushState() {
	const quick = pushState();
	if (quick !== "check") return quick;
	try {
		return await (await navigator.serviceWorker.getRegistration("/sw.js"))?.pushManager.getSubscription() && Notification.permission === "granted" ? "enabled" : "disabled";
	} catch {
		return "disabled";
	}
}
async function enablePush() {
	const quick = pushState();
	if (quick === "insecure") throw new Error("Οι ειδοποιήσεις χρειάζονται HTTPS. Το site τρέχει σε http.");
	if (quick === "unsupported") throw new Error("Ο browser δεν υποστηρίζει push ειδοποιήσεις.");
	if (quick === "denied") throw new Error("Οι ειδοποιήσεις είναι μπλοκαρισμένες. Επίτρεψέ τες από το εικονίδιο κλειδαριάς δίπλα στη διεύθυνση.");
	throw new Error("Λείπει το VITE_VAPID_PUBLIC_KEY.");
}
async function disablePush() {
	const sub = await (await navigator.serviceWorker.getRegistration("/sw.js"))?.pushManager.getSubscription();
	if (!sub) return;
	await supabase.rpc("unregister_push_subscription", { p_endpoint: sub.endpoint });
	await sub.unsubscribe();
}

async function invoke(name, body) {
	const { data, error } = await supabase.functions.invoke(name, { body });
	if (error) {
		const res = error.context;
		if (res && typeof res.json === "function") try {
			const payload = await res.json();
			if (payload?.error) throw new Error(ERROR_TEXT[payload.error] ?? payload.error);
		} catch (e) {
			if (e instanceof Error && e.message) throw e;
		}
		throw new Error(error.message || "Request failed.");
	}
	const payload = data;
	if (payload?.error) throw new Error(ERROR_TEXT[payload.error] ?? payload.error);
	return data;
}
