type FormState = {
  title: string;
  description: string;
  starts_at: string; // datetime-local
  duration_minutes: string;
  price: string; // euros
  capacity: string;
  published: boolean;
};

const empty: FormState = {
  title: "",
  description: "",
  starts_at: "",
  duration_minutes: "90",
  price: "",
  capacity: "",
  published: true,
};

function toLocalInput(iso: string) {
  const d = new Date(iso);
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

export function AdminOnlineSeminars() {
  const [reload, setReload] = useState(0);
  const seminars = useAsync(getAdminOnlineSeminars, [reload]);
  const bookings = useAsync(getOnlineBookings, [reload]);

  const [editing, setEditing] = useState<OnlineSeminar | null>(null);
  const [open, setOpen] = useState(false);
  const [form, setForm] = useState<FormState>(empty);
  const [saving, setSaving] = useState(false);
  const [msg, setMsg] = useState<string | null>(null);
  const [filter, setFilter] = useState<"paid" | "all">("paid");

  const refresh = () => setReload((n) => n + 1);

  function openNew() {
    setEditing(null);
    setForm(empty);
    setOpen(true);
  }
  function openEdit(s: OnlineSeminar) {
    setEditing(s);
    setForm({
      title: s.title,
      description: s.description,
      starts_at: toLocalInput(s.starts_at),
      duration_minutes: String(s.duration_minutes),
      price: (s.price_cents / 100).toString(),
      capacity: s.capacity?.toString() ?? "",
      published: s.published,
    });
    setOpen(true);
  }

  const valid =
    form.title.trim().length >= 3 && form.starts_at && Number(form.price) >= 0 && form.price !== "" &&
    Number(form.duration_minutes) > 0;

  async function submit() {
    if (!valid || saving) return;
    setSaving(true);
    setMsg(null);
    try {
      await saveOnlineSeminar(
        {
          title: form.title.trim(),
          description: form.description.trim(),
          starts_at: new Date(form.starts_at).toISOString(),
          duration_minutes: Number(form.duration_minutes),
          price_cents: Math.round(Number(form.price) * 100),
          currency: "eur",
          capacity: form.capacity ? Number(form.capacity) : null,
          published: form.published,
        },
        editing?.id,
      );
      setOpen(false);
      setMsg(editing ? "Το σεμινάριο ενημερώθηκε." : "Το σεμινάριο δημιουργήθηκε.");
      refresh();
    } catch (e) {
      setMsg("Αποτυχία: " + (e instanceof Error ? e.message : "άγνωστο σφάλμα"));
    } finally {
      setSaving(false);
    }
  }

  async function togglePublish(s: OnlineSeminar) {
    try {
      await setOnlineSeminarPublished(s.id, !s.published);
      refresh();
    } catch (e) {
      setMsg("Αποτυχία: " + (e instanceof Error ? e.message : "άγνωστο σφάλμα"));
    }
  }

  async function changeBooking(b: OnlineBooking, status: OnlineBooking["status"]) {
    const text =
      status === "paid"
        ? `Μαρκάρισμα του ${b.full_name} ως πληρωμένου;`
        : `Αναίρεση πληρωμής για τον ${b.full_name};`;
    if (!window.confirm(text)) return;
    try {
      await setOnlineBookingStatus(b.id, status);
      setMsg(status === "paid" ? `Ο ${b.full_name} μαρκαρίστηκε ως πληρωμένος.` : "Η πληρωμή αναιρέθηκε.");
      refresh();
    } catch (e) {
      setMsg("Αποτυχία: " + (e instanceof Error ? e.message : "άγνωστο σφάλμα"));

    >
      {children}
      {label}
    </button>
  );
}

function UserAvatar({
  src,
  alt,
  size = "h-9 w-9",
}: {
  src?: string | null;
  alt?: string | null;
  size?: string;
}) {
  const [error, setError] = useState(false);

  useEffect(() => {
    setError(false);
  }, [src]);

  const initials =
    (alt || "?")
      .trim()
      .split(" ")
      .map((p) => p[0])
      .filter(Boolean)
      .slice(0, 2)
      .join("")
      .toUpperCase() || "?";

  if (!src || error) {
    return (
      <span
        className={`flex ${size} items-center justify-center rounded-full border border-white/20 bg-white/10 text-xs font-medium text-white/80 shadow-[0_0_12px_rgba(255,255,255,0.1)] transition-transform duration-300 group-hover:scale-105 group-hover:border-white/50`}
      >
        {initials}
      </span>
    );
  }

  return (
    <img
      src={src}
      alt={alt || "Avatar"}
      onError={() => setError(true)}
      className={`${size} rounded-full border border-white/20 object-cover shadow-[0_0_12px_rgba(255,255,255,0.1)] transition-transform duration-300 group-hover:scale-105 group-hover:border-white/50`}
    />
  );
}

export interface OnlineCheckoutResult {
  /** Viva hosted Smart Checkout URL. */
  redirectUrl: string;

/**
 * Only allow redirects to the official Viva Smart Checkout hosts.
 * This prevents an unexpected URL from the Edge Function being trusted.
 */
function isValidVivaCheckoutUrl(value: unknown): value is string {
  if (typeof value !== "string") return false;

  try {
    const url = new URL(value);

    const allowedHosts = [
      "demo.vivapayments.com",
      "www.vivapayments.com",
    ];

    return (
      url.protocol === "https:" &&
      allowedHosts.includes(url.hostname) &&
      url.pathname === "/web/checkout" &&
      /^\d{6,20}$/.test(url.searchParams.get("ref") ?? "")
    );
  } catch {
    return false;
  }
}

export async function startOnlineCheckout(p: {

  }

  // Send the authenticated user's token when available.
  // The Edge Function verifies the user and calculates any member discount.
  const { data: sessionData, error: sessionError } =

  }

        },

          fullName: p.fullName.trim(),
          email: p.email.trim(),
          phone: p.phone.trim(),
        }),
      },
    );
  } catch {
    throw new Error("network_error");
  }

  const data = await response.json().catch(() => null);

  if (!response.ok) {
    console.error("create-online-checkout failed:", response.status, data);

    throw new Error("invalid_viva_checkout_response");
  }

  return {
    redirectUrl: data.redirectUrl,
    bookingId: data.bookingId,
    amountCents: data.amountCents,
    listPriceCents: data.listPriceCents,

/**
 * Polls AURA until the Viva webhook marks this booking as paid.
 *
 * IMPORTANT:
 * The get_online_booking_status RPC must accept p_booking_id
 * and return the booking status. It must no longer search by Stripe session ID.
 */
export async function waitForPaid(
  bookingId: string,
  tries = 20,
  delayMs = 2000,
): Promise<boolean> {
  if (!bookingId) {
    throw new Error("booking_id_required");
  }

  for (let i = 0; i < tries; i++) {
    const { data, error } = await supabase.rpc(
      "get_online_booking_status",
      { p_booking_id: bookingId },
    );

    if (error) {
      console.error("get_online_booking_status failed:", error);
      throw new Error("booking_status_check_failed");
    }

    if (data === "paid") return true;
    if (data === "failed" || data === "refunded") return false;

    if (i < tries - 1) {
      await new Promise((resolve) => setTimeout(resolve, delayMs));
    }
  }

  return false;
}
