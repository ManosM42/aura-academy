/** Metallic silver text (vertical chrome, with a soft glow). */
const silverText = {
  backgroundImage: CHROME_FILL,
  WebkitBackgroundClip: "text",
  backgroundClip: "text",
  WebkitTextFillColor: "transparent",
} as const;

/** "AURA ELITE | -25%" pill with a slow light sweep. */
export function MemberBadge({
  planId,
  percent,
  className = "",
}: {
  planId: PlanId | null;
  percent: number;
  className?: string;
}) {
  const reduce = useReducedMotion();
  if (percent <= 0) return null;
  const Icon = planId === "high" ? Crown : BadgeCheck;
  const name = getPlan(planId)?.name ?? "AURA MEMBER";

  return (
    <span
      className={`relative inline-flex items-center gap-1.5 overflow-hidden rounded-full border border-white/60 px-3 py-1 text-[10px] font-extrabold uppercase tracking-[0.18em] text-neutral-950 shadow-[0_0_26px_rgba(255,255,255,0.28),inset_0_1px_0_rgba(255,255,255,0.95)] ${className}`}
      style={{ backgroundImage: CHROME_FILL }}
    >
      <Icon className="relative size-3" strokeWidth={2.6} aria-hidden />
      <span className="relative">{name}</span>
      <span aria-hidden className="relative h-3 w-px bg-black/25" />
      <span className="relative">−{percent}%</span>
      {!reduce && (
        <motion.span
          aria-hidden
          className="absolute inset-y-0 -left-1/2 w-1/2 -skew-x-12 bg-white/80 blur-md"
          animate={{ x: ["0%", "420%"] }}
          transition={{ duration: 2.4, repeat: Infinity, repeatDelay: 3.4, ease: "easeInOut" }}
        />
      )}
    </span>
  );
}

const SIZES = {
  sm: { list: "text-[11px]", final: "text-base" },
  md: { list: "text-xs", final: "text-2xl" },
  lg: { list: "text-sm", final: "text-4xl" },
} as const;

/**
 * Shows the list price struck through and the member price in silver.
 * With no discount it renders just the price (no strike-through).
 */
export function DiscountedPrice({
  listCents,
  finalCents,
  format,
  size = "md",
  align = "end",
  layout = "stacked",
  className = "",
}: {
  listCents: number;
  finalCents: number;
  format: (cents: number) => string;
  size?: keyof typeof SIZES;
  align?: "start" | "end";
  layout?: "stacked" | "inline";
  className?: string;
}) {
  const s = SIZES[size];
  const discounted = finalCents < listCents;

  const final = (
    <span
      className={`${s.final} font-extrabold tracking-tight drop-shadow-[0_0_18px_rgba(255,255,255,0.45)]`}
      style={silverText}
    >
      {format(finalCents)}
    </span>
  );

  if (!discounted) {
    return <span className={`inline-block ${className}`}>{final}</span>;
  }

  const list = (
    <s className={`${s.list} font-medium text-zinc-500 decoration-zinc-500/80`} aria-label={`Initial price ${format(listCents)}`}>
      {format(listCents)}
    </s>
  );

  if (layout === "inline") {
    return (
      <span className={`inline-flex items-baseline gap-2 ${className}`}>
        {list}
        {final}
      </span>
    );
  }

  return (
    <span
      className={`inline-flex flex-col leading-none ${align === "end" ? "items-end text-right" : "items-start text-left"} ${className}`}
    >
      {list}
      <span className="mt-1.5">{final}</span>
    </span>
  );
}

import { useEffect, useState } from "react";
import { signedMediaUrl } from "@/lib/queries";

const VIDEO_RE = /\.(mp4|mov|webm|m4v)$/i;

export function MediaThumb({ path }: { path: string }) {
  const [url, setUrl] = useState<string | null>(null);
  const [error, setError] = useState(false);

  useEffect(() => {
    let active = true;
    setUrl(null);
    setError(false);
    signedMediaUrl(path)
      .then((u) => active && setUrl(u))
      .catch(() => active && setError(true));
    return () => {
      active = false;
    };
  }, [path]);

  if (error) {
    return (
      <div className="flex h-28 w-full items-center justify-center rounded-lg border border-red-500/30 bg-red-500/5 text-xs text-red-300">
        Αποτυχία φόρτωσης
      </div>
    );
  }

  if (!url) {
    return <div className="h-28 w-full animate-pulse rounded-lg bg-white/5" />;
  }

  if (VIDEO_RE.test(path)) {
    return (
      <video
        src={url}
        controls
        className="h-28 w-full rounded-lg bg-black object-cover"
      />
    );
  }

  return (
    <img
      src={url}
      alt="Submission media"
      className="h-28 w-full rounded-lg object-cover"
      loading="lazy"
    />
  );
}

import { useEffect, useRef, useState, type DragEvent } from "react";
import { motion } from "motion/react";
import { ImagePlus, Loader2, X } from "lucide-react";
import { CAPTION_MAX, publishPost, type PortfolioPost } from "@/lib/portfolio";

const MAX_INPUT_BYTES = 25 * 1024 * 1024;
const ACCEPT = "image/jpeg,image/png,image/webp,image/heic,image/heif";

export default function NewPostModal({
  userId,
  onClose,
  onPublished,
}: {
  userId: string;
  onClose: () => void;
  onPublished: (post: PortfolioPost) => void;
}) {
  const [file, setFile] = useState<File | null>(null);
  const [preview, setPreview] = useState<string | null>(null);
  const [caption, setCaption] = useState("");
  const [busy, setBusy] = useState(false);
  const [dragging, setDragging] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (!file) {
      setPreview(null);
      return;
    }
    const url = URL.createObjectURL(file);
    setPreview(url);
    return () => URL.revokeObjectURL(url);
  }, [file]);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && !busy && onClose();
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [onClose, busy]);

  function pick(f: File | undefined | null) {
    setError(null);
    if (!f) return;
    if (!f.type.startsWith("image/")) {
      setError("Διάλεξε αρχείο εικόνας (JPG, PNG ή WebP).");
      return;
    }
    if (f.size > MAX_INPUT_BYTES) {
      setError("Η φωτογραφία είναι πολύ μεγάλη (έως 25 MB).");
      return;
    }
    setFile(f);
  }

  function onDrop(e: DragEvent<HTMLDivElement>) {
    e.preventDefault();
    setDragging(false);
    pick(e.dataTransfer.files?.[0]);
  }

  async function publish() {
    if (!file || busy) return;
    setBusy(true);
    setError(null);
    try {
      const post = await publishPost(userId, file, caption);
      onPublished(post);
    } catch (e) {
      setError(e instanceof Error ? e.message : "Το post δεν δημοσιεύτηκε.");
      setBusy(false);
    }
  }

  return (
    <div className="pf-modal-layer" onPointerDown={() => !busy && onClose()}>
      <motion.div
        role="dialog"
        aria-modal="true"
        aria-labelledby="pf-newpost-title"
        initial={{ opacity: 0, y: 24, scale: 0.97 }}
        animate={{ opacity: 1, y: 0, scale: 1 }}
        transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
        className="pf-modal pf-modal-wide"
        onPointerDown={(e) => e.stopPropagation()}
      >
        <div className="flex items-center justify-between">
          <h2 id="pf-newpost-title" className="pf-title text-xl font-bold tracking-tight">
            Νέο post
          </h2>
          <button type="button" className="pf-iconbtn" style={{ width: 34, height: 34 }} onClick={onClose} disabled={busy} aria-label="Κλείσιμο">
            <X className="size-4" />
          </button>
        </div>

        <input ref={inputRef} type="file" accept={ACCEPT} hidden onChange={(e) => pick(e.target.files?.[0])} />

        <div className="mt-5">
          {preview ? (
            <div className="relative overflow-hidden rounded-2xl border border-white/10 bg-black">
              <img src={preview} alt="Προεπισκόπηση" className="mx-auto max-h-[46vh] w-auto object-contain" />
              <button
                type="button"
                className="pf-btn absolute bottom-3 right-3"
                onClick={() => inputRef.current?.click()}
                disabled={busy}
              >
                Αλλαγή
              </button>
            </div>
          ) : (
            <div
              role="button"
              tabIndex={0}
              className={`pf-drop ${dragging ? "pf-drop-on" : ""}`}
              onClick={() => inputRef.current?.click()}
              onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && inputRef.current?.click()}
              onDragOver={(e) => {
                e.preventDefault();
                setDragging(true);
              }}
              onDragLeave={() => setDragging(false)}
              onDrop={onDrop}
            >
              <div>
                <ImagePlus className="mx-auto size-9 text-white/70" />
                <p className="mt-3 text-sm text-white/80">Σύρε εδώ μια φωτογραφία ή πάτα για επιλογή</p>
                <p className="mt-1 text-xs text-white/40">JPG, PNG ή WebP · θα βελτιστοποιηθεί αυτόματα</p>
              </div>
            </div>
          )}
        </div>

        <label className="mt-5 block">
          <span className="mb-1.5 flex items-center justify-between text-[11px] uppercase tracking-[0.22em] text-white/45">
            Περιγραφή
            <span className="normal-case tracking-normal">
              {caption.length}/{CAPTION_MAX}
            </span>
          </span>
          <textarea
            className="pf-input min-h-[96px] resize-y"
            placeholder="Πες λίγα λόγια: τεχνική, εργαλεία, ιστορία του κουρέματος…"
            value={caption}
            maxLength={CAPTION_MAX}
            onChange={(e) => setCaption(e.target.value)}
            disabled={busy}
          />
        </label>

        {error && (
          <p className="pf-toast mt-4" role="alert">
            {error}
          </p>
        )}

        <div className="mt-6 flex gap-3">
          <button type="button" className="pf-btn pf-btn-solid flex-1 py-3" onClick={() => void publish()} disabled={!file || busy}>
            {busy ? <Loader2 className="size-4 animate-spin" /> : null}
            {busy ? "Δημοσίευση…" : "ΔΗΜΟΣΙΕΥΣΗ"}
          </button>
          <button type="button" className="pf-btn" onClick={onClose} disabled={busy}>
            Άκυρο
          </button>

    </div>
  );

import { useEffect, useRef, useState } from "react";
import { motion, AnimatePresence } from "motion/react";
import { Trash2, Volume2, VolumeX, X } from "lucide-react";
import { markStoryViewed, deleteStory, type StoryGroup } from "@/lib/queries";

const STORY_DURATION_MS = 5000;

export default function StoryViewer({
  groups,
  startGroupIndex,
  currentUserId,
  onClose,
  onDeleted,
}: {
  groups: StoryGroup[];
  startGroupIndex: number;
  currentUserId: string;
  onClose: () => void;
  onDeleted: () => void;
}) {
  const [groupIndex, setGroupIndex] = useState(startGroupIndex);
  const [storyIndex, setStoryIndex] = useState(0);
  const [progress, setProgress] = useState(0);
  const [paused, setPaused] = useState(false);
  const [muted, setMuted] = useState(false);
  const videoRef = useRef<HTMLVideoElement>(null);
  const audioRef = useRef<HTMLAudioElement>(null);
  const rafRef = useRef<number>();
  const startRef = useRef<number>(0);
  const viewedRef = useRef<Set<string>>(new Set());

  const group = groups[groupIndex];
  const story = group?.stories[storyIndex];

  function goNextStory() {
    if (!group) return;
    if (storyIndex < group.stories.length - 1) {
      setStoryIndex((i) => i + 1);
    } else if (groupIndex < groups.length - 1) {
      setGroupIndex((g) => g + 1);
      setStoryIndex(0);
    } else {
      onClose();
    }
  }

  function goPrevStory() {
    if (storyIndex > 0) {
      setStoryIndex((i) => i - 1);
    } else if (groupIndex > 0) {
      const prevGroup = groups[groupIndex - 1];
      setGroupIndex((g) => g - 1);
      setStoryIndex(prevGroup.stories.length - 1);
    }
  }

  // Mark as viewed once, drive the progress bar, and (re)start any attached audio.
  useEffect(() => {
    if (!story) return;
    if (!viewedRef.current.has(story.id)) {
      viewedRef.current.add(story.id);
      markStoryViewed(story.id).catch(() => {});
    }

    setProgress(0);
    startRef.current = performance.now();

    if (audioRef.current) {
      audioRef.current.pause();
      audioRef.current.currentTime = 0;
      if (story.audio_url) {
        audioRef.current.src = story.audio_url;
        audioRef.current.muted = muted;
        audioRef.current.loop = true;
        audioRef.current.play().catch(() => {});
      }
    }

    const isVideo = story.media_type === "video";
    if (isVideo) return; // videos drive their own progress via timeupdate

    function tick(now: number) {
      if (paused) {
        startRef.current = now - (progress / 100) * STORY_DURATION_MS;
        rafRef.current = requestAnimationFrame(tick);
        return;
      }
      const elapsed = now - startRef.current;
      const pct = Math.min(100, (elapsed / STORY_DURATION_MS) * 100);
      setProgress(pct);
      if (pct >= 100) {
        goNextStory();
        return;
      }
      rafRef.current = requestAnimationFrame(tick);
    }
    rafRef.current = requestAnimationFrame(tick);
    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [story?.id, paused]);

  // Stop any playing audio on unmount (closing the viewer).
  useEffect(() => {
    return () => {
      audioRef.current?.pause();
    };
  }, []);

  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if (e.key === "Escape") onClose();
      if (e.key === "ArrowRight") goNextStory();
      if (e.key === "ArrowLeft") goPrevStory();
    }
    window.addEventListener("keydown", handleKey);
    return () => window.removeEventListener("keydown", handleKey);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [groupIndex, storyIndex]);

  function toggleMute() {
    setMuted((m) => {
      const next = !m;
      if (audioRef.current) audioRef.current.muted = next;
      if (videoRef.current) videoRef.current.muted = next;
      return next;
    });
  }

  async function handleDelete() {
    if (!story) return;
    await deleteStory(story.id);
    onDeleted();
    goNextStory();
  }

  if (!group || !story) return null;
  const isOwn = group.author.id === currentUserId;

  return (
    <AnimatePresence>
      <motion.div
        initial={{ opacity: 0 }}
        animate={{ opacity: 1 }}
        exit={{ opacity: 0 }}
        className="fixed inset-0 z-[80] flex items-center justify-center bg-black"
      >
        <motion.div
          drag="y"
          dragConstraints={{ top: 0, bottom: 0 }}
          dragElastic={0.6}
          onDragEnd={(_, info) => {
            if (info.offset.y > 100) onClose();
          }}
          className="relative h-full w-full max-w-md overflow-hidden bg-black sm:h-[92vh] sm:rounded-2xl"
        >
          {/* progress bars */}
          <div className="absolute inset-x-0 top-0 z-10 flex gap-1 p-2">
            {group.stories.map((s, i) => (
              <div key={s.id} className="h-0.5 flex-1 overflow-hidden rounded-full bg-white/30">
                <div
                  className="h-full bg-white"
                  style={{
                    width: i < storyIndex ? "100%" : i === storyIndex ? `${progress}%` : "0%",
                  }}
                />
              </div>
            ))}
          </div>

          {/* header */}
          <div className="absolute inset-x-0 top-4 z-10 flex items-center justify-between px-3">
            <div className="flex items-center gap-2">
              {group.author.avatar_url ? (
                <img src={group.author.avatar_url} className="h-8 w-8 rounded-full object-cover ring-1 ring-white/30" alt="" />
              ) : (
                <div className="flex h-8 w-8 items-center justify-center rounded-full bg-white/20 text-xs font-semibold text-white">
                  {group.author.full_name?.[0]?.toUpperCase() ?? "A"}
                </div>
              )}
              <span className="text-sm font-semibold text-white drop-shadow">
                {group.author.full_name ?? "Χρήστης"}
              </span>
            </div>
            <div className="flex items-center gap-1">
              {(story.audio_url || story.media_type === "video") && (
                <button
                  onClick={toggleMute}
                  className="rounded-full p-2 text-white/80 hover:bg-white/10 hover:text-white"
                >
                  {muted ? <VolumeX size={18} /> : <Volume2 size={18} />}
                </button>
              )}
              {isOwn && (
                <button
                  onClick={handleDelete}
                  className="rounded-full p-2 text-white/80 hover:bg-white/10 hover:text-white"
                >
                  <Trash2 size={18} />
                </button>
              )}
              <button
                onClick={onClose}
                className="rounded-full p-2 text-white/80 hover:bg-white/10 hover:text-white"
              >
                <X size={20} />
              </button>
            </div>
          </div>

          {/* media */}
          <div className="flex h-full w-full items-center justify-center">
            {story.media_type === "video" ? (
              <video
                ref={videoRef}
                src={story.media_url}
                className="h-full w-full object-contain"
                autoPlay
                muted={muted}
                playsInline
                onTimeUpdate={(e) => {
                  const v = e.currentTarget;
                  if (v.duration) setProgress((v.currentTime / v.duration) * 100);
                }}
                onEnded={goNextStory}
              />
            ) : (
              <img src={story.media_url} className="h-full w-full object-contain" alt="" />
            )}
          </div>

          {/* attached audio (image stories only — video has its own soundtrack) */}
          <audio ref={audioRef} className="hidden" />

          {story.audio_title && (
            <div className="absolute bottom-4 left-1/2 z-10 flex -translate-x-1/2 items-center gap-2 rounded-full bg-black/50 px-3 py-1.5 text-xs text-white backdrop-blur">
              🎵 <span className="max-w-[160px] truncate">{story.audio_title}</span>
            </div>
          )}

          {/* tap zones */}
          <button
            aria-label="Previous"
            onClick={goPrevStory}
            onMouseDown={() => setPaused(true)}
            onMouseUp={() => setPaused(false)}
            onTouchStart={() => setPaused(true)}
            onTouchEnd={() => setPaused(false)}
            className="absolute inset-y-0 left-0 z-[5] w-1/3"
          />
          <button
            aria-label="Next"
            onClick={goNextStory}
            onMouseDown={() => setPaused(true)}
            onMouseUp={() => setPaused(false)}
            onTouchStart={() => setPaused(true)}
            onTouchEnd={() => setPaused(false)}
            className="absolute inset-y-0 right-0 z-[5] w-2/3"
          />
        </motion.div>
      </motion.div>
    </AnimatePresence>
  );
}

// ⚙️ Το edge function aura-tts (Gemini TTS). Προσαρμόσε αν τα env σου λέγονται αλλιώς.
const TTS_URL = `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/aura-tts`;
const TTS_KEY: string =
  import.meta.env.VITE_SUPABASE_PUBLISHABLE_KEY ?? import.meta.env.VITE_SUPABASE_ANON_KEY ?? "";
const TTS_TIMEOUT_MS = 15000;

export type VoiceIssue = null | "no-greek-voice" | "no-audio";

const hasGreek = (t: string) => /[\u0370-\u03FF\u1F00-\u1FFF]/.test(t);
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

// ── Ήχος: ένα μόνιμο <audio> που «ξεκλειδώνει» με την πρώτη χειρονομία (iOS) ──
const SILENT_WAV = "data:audio/wav;base64,UklGRiQAAABXQVZFZm10IBAAAAABAAEAQB8AAEAfAAABAAgAZGF0YQAAAAA=";
let audioEl: HTMLAudioElement | null = null;
const getAudio = () => {
  if (!audioEl) {
    audioEl = new Audio();
    audioEl.preload = "auto";
    (audioEl as any).playsInline = true;
  }
  return audioEl;
};
const unlockAudio = () => {
  const a = getAudio();
  if (a.src) return;
  a.src = SILENT_WAV;
  a.play()
    .then(() => {
      if (a.src.startsWith("data:")) a.pause();
    })
    .catch(() => {});
};

// ── Ανάλυση WAV: περιβάλλουσα έντασης (για να κλείνει το στόμα στις παύσεις) ──
const FRAME = 0.02; // 20ms
type Clip = { url: string; env: Float32Array; start: number; end: number };

function analyseWav(buf: ArrayBuffer): Omit<Clip, "url"> {
  const dv = new DataView(buf);
  const sr = dv.getUint32(24, true) || 24000;
  const n = Math.floor((buf.byteLength - 44) / 2);
  const per = Math.max(1, Math.round(sr * FRAME));
  const frames = Math.max(1, Math.ceil(n / per));
  const env = new Float32Array(frames);

  for (let f = 0; f < frames; f++) {
    let sum = 0;
    let cnt = 0;
    for (let i = f * per; i < Math.min(n, (f + 1) * per); i++, cnt++) {
      const x = dv.getInt16(44 + i * 2, true) / 32768;
      sum += x * x;
    }
    env[f] = cnt ? Math.sqrt(sum / cnt) : 0;
  }

  const sorted = Array.from(env).sort((a, b) => a - b);
  const ref = Math.max(1e-4, sorted[Math.floor(sorted.length * 0.95)]);
  for (let f = 0; f < frames; f++) env[f] = Math.min(1, env[f] / ref);
  // ελαφρύ εξομάλυνση 3 καρέ
  const sm = new Float32Array(frames);
  for (let f = 0; f < frames; f++) {
    sm[f] = ((env[f - 1] ?? env[f]) + env[f] * 2 + (env[f + 1] ?? env[f])) / 4;
  }

  let a = 0;
  let b = frames - 1;
  while (a < frames - 1 && sm[a] < 0.12) a++;
  while (b > a && sm[b] < 0.12) b--;
  return { env: sm, start: a * FRAME, end: (b + 1) * FRAME };
}

const clipCache = new Map<string, Clip>();
async function fetchClip(text: string): Promise<Clip> {
  const hit = clipCache.get(text);
  if (hit) return hit;

  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), TTS_TIMEOUT_MS);
  try {
    const res = await fetch(TTS_URL, {
      method: "POST",
      signal: ctrl.signal,
      headers: { "Content-Type": "application/json", apikey: TTS_KEY, Authorization: `[REDACTED]` },
      body: JSON.stringify({ text }),
    });
    if (!res.ok || !(res.headers.get("content-type") ?? "").includes("audio")) throw new Error("tts-unavailable");
    const buf = await res.arrayBuffer();
    const clip: Clip = {
      ...analyseWav(buf),
      url: URL.createObjectURL(new Blob([buf], { type: "audio/wav" })),
    };
    if (clipCache.size >= 20) {
      const k = clipCache.keys().next().value!;
      URL.revokeObjectURL(clipCache.get(k)!.url);
      clipCache.delete(k);
    }
    clipCache.set(text, clip);
    return clip;
  } finally {
    clearTimeout(timer);
  }
}

function loadVoices(): Promise<SpeechSynthesisVoice[]> {
  const synth = window.speechSynthesis;
  const now = synth.getVoices();
  if (now.length) return Promise.resolve(now);
  return new Promise((resolve) => {
    const done = () => {
      synth.removeEventListener("voiceschanged", done);
      resolve(synth.getVoices());
    };
    synth.addEventListener("voiceschanged", done);
    setTimeout(done, 1000);
  });
}

export function useAuraSpeech() {
  const [speaking, setSpeaking] = useState(false);
  const [voiceIssue, setVoiceIssue] = useState<VoiceIssue>(null);
  const st = useRef({
    text: "",
    visemes: [] as Viseme[],
    pos: 0,
    lastTick: 0,
    rate: 1,
    active: false,
    hasBoundary: false,
    token: 0,
    clip: null as Clip | null,
  });
  const uttRef = useRef<SpeechSynthesisUtterance | null>(null); // κρατάμε αναφορά (το Chrome κάνει GC στα utterances)
  const timers = useRef<number[]>([]);

  const clearTimers = useCallback(() => {
    timers.current.forEach((t) => clearTimeout(t));
    timers.current = [];
  }, []);

  const finish = useCallback(() => {
    st.current.active = false;
    st.current.clip = null;
    setSpeaking(false);
    clearTimers();
  }, [clearTimers]);

  const stop = useCallback(() => {
    st.current.token++;
    if (typeof window !== "undefined") {
      if ("speechSynthesis" in window) window.speechSynthesis.cancel();
      if (audioEl && !audioEl.src.startsWith("data:")) audioEl.pause();
    }
    finish();
  }, [finish]);

  useEffect(() => () => stop(), [stop]);

  // ξεκλείδωμα ήχου στην πρώτη χειρονομία του χρήστη (απαραίτητο σε iOS/Safari)
  useEffect(() => {
    const events = ["pointerdown", "touchend", "keydown", "click"] as const;
    const h = () => {
      unlockAudio();
      events.forEach((e) => window.removeEventListener(e, h));
    };
    events.forEach((e) => window.addEventListener(e, h, { passive: true }));
    return () => events.forEach((e) => window.removeEventListener(e, h));
  }, []);

  const speak = useCallback(
    async (text: string) => {
      if (typeof window === "undefined") return;
      const s = st.current;
      const token = ++s.token;
      clearTimers();
      s.text = text;
      s.visemes = visemesForText(text);
      s.pos = 0;
      s.hasBoundary = false;
      s.active = false;
      s.clip = null;
      s.rate = 1;
      setVoiceIssue(null);
      if (audioEl) audioEl.pause();
      if ("speechSynthesis" in window) window.speechSynthesis.cancel();

      // Το avatar κινείται και χωρίς ήχο (όταν δεν υπάρχει φωνή)
      const startSilent = () => {
        if (s.token !== token) return;
        s.hasBoundary = false;
        s.clip = null;
        s.pos = 0;
        s.lastTick = performance.now();
        s.active = true;
        setSpeaking(true);
        const ms = (text.length / CHARS_PER_SEC) * 1000 + 300;
        timers.current.push(window.setTimeout(() => s.token === token && finish(), ms));
      };

      // ── Φωνή της συσκευής (εφεδρική) ─────────────────────────────────────
      const speakWithBrowser = async () => {
        if (!("speechSynthesis" in window)) {
          setVoiceIssue("no-audio");
          startSilent();
          return;
        }
        const synth = window.speechSynthesis;
        synth.cancel();

        const greek = hasGreek(text);
        const voices = await loadVoices();
        if (s.token !== token) return;
        const voice = voices.find((v) =>
          v.lang.toLowerCase().replace("_", "-").startsWith(greek ? "el" : "en"),
        );
        if (greek && !voice) {
          setVoiceIssue("no-greek-voice");
          startSilent();
          return;
        }

        await sleep(80); // Chrome: speak() αμέσως μετά το cancel() χάνεται
        if (s.token !== token) return;

        const u = new SpeechSynthesisUtterance(text);
        u.lang = greek ? "el-GR" : "en-US";
        if (voice) u.voice = voice;
        u.rate = s.rate;

        u.onstart = () => {
          if (s.token !== token) return;
          s.active = true;
          s.pos = 0;
          s.lastTick = performance.now();
          setSpeaking(true);
        };
        u.onboundary = (e) => {
          if (s.token !== token) return;
          s.hasBoundary = true;
          s.pos = e.charIndex; // ξανασυγχρονισμός στην αρχή της λέξης
        };
        u.onend = () => {
          if (s.token === token) finish();
        };
        u.onerror = (e) => {
          if (s.token !== token) return;
          if (e.error === "canceled" || e.error === "interrupted") return;
          setVoiceIssue("no-audio");
          if (!s.active) startSilent();
          else finish();
        };

        uttRef.current = u;
        synth.resume();
        synth.speak(u);

        // Αν σε 2s δεν ξεκίνησε ήχος, πέφτουμε σε σιωπηλό animation
        timers.current.push(
          window.setTimeout(() => {
            if (s.token === token && !s.active) {
              synth.cancel();
              setVoiceIssue("no-audio");
              startSilent();
            }
          }, 2000),
        );
      };

      // ── Κύρια φωνή: ρεαλιστική φωνή από το edge function ────────────────
      let clip: Clip | null = null;
      try {
        clip = await fetchClip(text);
      } catch (e) {
        console.warn("aura-tts:", String(e));
      }
      if (s.token !== token) return;
      if (!clip) return speakWithBrowser();

      const a = getAudio();
      a.onplaying = () => {
        if (s.token !== token) return;
        s.clip = clip;
        s.active = true;
        setSpeaking(true);
      };
      a.onended = () => {
        if (s.token === token) finish();
      };
      a.onerror = () => {
        if (s.token === token) {
          finish();
          void speakWithBrowser();
        }
      };
      a.src = clip.url;
      a.currentTime = 0;
      try {
        await a.play();
      } catch {
        // αυτόματη αναπαραγωγή μπλοκαρίστηκε από τον browser
        if (s.token === token) {
          setVoiceIssue("no-audio");
          startSilent();
        }
      }
    },
    [finish, clearTimers],
  );

  /** Καλείται σε κάθε frame από το avatar. */
  const getViseme = useCallback((now: number): Viseme => {
    const s = st.current;
    if (!s.active) return CLOSED;

    // Πραγματικός ήχος: χρόνος αναπαραγωγής → γράμμα, με ένταση που ανοίγει/κλείνει το στόμα
    if (s.clip) {
      const a = audioEl;
      if (!a || a.paused) return CLOSED;
      const t = a.currentTime;
      const c = s.clip;
      const level = c.env[Math.min(c.env.length - 1, Math.floor(t / FRAME))] ?? 0;
      const prog = Math.min(1, Math.max(0, (t - c.start) / Math.max(0.2, c.end - c.start)));
      const idx = Math.min(s.visemes.length - 1, Math.floor(prog * s.visemes.length));
      const v = s.visemes[idx] ?? CLOSED;
      const gate = Math.min(1, level * 1.5);
      return gate < 0.06 ? CLOSED : ({ ...v, open: v.open * (0.35 + 0.65 * gate) } as Viseme);
    }

    const dt = (now - s.lastTick) / 1000;
    s.lastTick = now;

    const ch = s.text[Math.floor(s.pos)] ?? "";
    const hold = s.hasBoundary && /[\s.,;:!?·…]/.test(ch);
    if (!hold) s.pos += dt * CHARS_PER_SEC * s.rate;

    const idx = Math.floor(s.pos);
    return idx >= s.visemes.length ? CLOSED : (s.visemes[idx] ?? CLOSED);
  }, []);

  return { speak, stop, speaking, voiceIssue, getViseme };
}
