function timeAgo(iso) {
	const diffMs = Date.now() - new Date(iso).getTime();
	const mins = Math.floor(diffMs / 6e4);
	if (mins < 1) return "τώρα";

function useOnlinePresence(me) {
	const [users, setUsers] = (0, import_react.useState)([]);
	(0, import_react.useEffect)(() => {
		if (!me) {
			setUsers([]);
			return;
		}
		const channel = supabase.channel(CHANNEL_NAME, { config: { presence: { key: me.id } } });
		channel.on("presence", { event: "sync" }, () => {
			const state = channel.presenceState();
			const flat = Object.values(state).map((presences) => presences[0]).filter(Boolean);
			setUsers(flat);
		});
		channel.subscribe(async (status) => {
			if (status === "SUBSCRIBED") await channel.track({
				id: me.id,
				full_name: me.full_name,
				avatar_url: me.avatar_url,
				role: me.role
			});
		});
		return () => {
			supabase.removeChannel(channel);
		};
	}, [
		me?.id,
		me?.full_name,
		me?.avatar_url,
		me?.role
	]);
	return users;
}
